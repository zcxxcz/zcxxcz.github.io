三条通用教训（都在这套题里踩到了）
  1. 贪心的单调性一定要证。T1 里「取最小的 k 个最省」看着很对，实际错在「cost 对右端点不单调」， 样例 2 的重复值 2 2 就能卡死它。写之前先想清楚「谁是单调的」。
  2. 「至少 1 个点」这类边界单独处理。T4 的剥叶核心可能为空，返回 0 会低估点数直接判错。
  3. 部分分不是白送的。T3 的三状态、T2 的线路图，都是从特殊性质里长出来的；先写特殊性质，再想怎么合并。

T1能量水晶 crystal

时限 1.0s内存 1024MiB20 个测试点，每点 5 分目录 crystal

题意与建模

给 n 个整数，最多做 m 次「把某个数 +1」，问最后最多有几个数相等。

两次观察就能把题做完：

  1. 目标值一定取某个水晶的原值。把目标设得比集合里的最大值还大，只会让每一颗都多花代价，没有任何好处。
  2. 排序后，选定「最大值」a[r]，要选的一定是它左边紧挨着的一段。因为代价是 a[r]-a[i]， 离 a[r] 越近越便宜；跳过近的选远的，把远的换成近的只会更省。

于是问题变成：排序后找一段最长的连续区间 [r-k+1, r]，使 cost(r,k) = k·a[r] − (pre[r] − pre[r−k]) ≤ m。

这里有个常见的坑（我第一次写就错了）：「固定颗数 k 时取最小的 k 个最省」是错的。 例如 a = [−3,−2,1,3,5,5,6]、m = 0：取最小的 2 个是 {−3,−2}，代价 1 超预算； 而两个 5 本身代价就是 0，答案是 2。 cost(r,k) 对 k 单调不增（所以能二分），但对 r 不单调，不能一上来就取 r=k。

10 分枚举子集测试点 1,2 ｜ O(2ⁿ·n)

直接枚举「哪些水晶参与合成」，把这批全部抬到它们的最大值，看总代价是否 ≤ m。 由于目标值取集合最大值最优，不用枚举目标值本身。

// crystal 部分分 1：n <= 5 的暴力（枚举子集）
// 期望得分：测试点 1,2（20 分）
#include <bits/stdc++.h>
using namespace std;
int n; long long m; long long a[10];
int main() {
    freopen("crystal.in", "r", stdin);
    freopen("crystal.out", "w", stdout);
    scanf("%d %lld", &n, &m);
    for (int i = 0; i < n; i++) scanf("%lld", &a[i]);
    int ans = 1;
    for (int mask = 1; mask < (1 << n); mask++) {
        long long mx = LLONG_MIN, sum = 0; int cnt = 0;
        for (int i = 0; i < n; i++) if (mask >> i & 1) {
            mx = max(mx, a[i]); sum += a[i]; cnt++;
        }
        long long cost = (long long)cnt * mx - sum;  // 全部抬到组内最大值
        if (cost <= m) ans = max(ans, cnt);
    }
    printf("%d\n", ans);
    return 0;
}
45 分排序后固定最大值向前扫测试点 1~9 ｜ O(n log n + n²)

排序后枚举右端点 r（即目标值 a[r]），从 a[r] 往前一个个加进来， 边走边累加代价 cost += a[r] − a[r−k+1]。由于固定 r 时代价随 k 单调不降， 一旦超过 m 就能直接 break，常数很小，n=5000 跑得飞快。

// crystal 部分分 2：排序后枚举“最大值”，逐个往前加，O(n^2)
// 期望得分：测试点 1~9（45 分）
#include <bits/stdc++.h>
using namespace std;
int n; long long m;
int main() {
    freopen("crystal.in", "r", stdin);
    freopen("crystal.out", "w", stdout);

    vector<long long> a(n + 1);
    for (int i = 1; i <= n; i++) scanf("%lld", &a[i]);
    sort(a.begin() + 1, a.end());
    int ans = 1;
    for (int r = 1; r <= n; r++) {              // 目标值 = a[r]
        long long cost = 0;
        for (int k = 1; k <= r; k++) {          // 取 a[r-k+1..r] 这 k 个
            cost += a[r] - a[r - k + 1];        // 第 k 个水晶的代价
            if (cost <= m) ans = max(ans, k);
            else break;                          // 代价单调不降，可以直接 break
        }
    }
    printf("%d\n", ans);
    return 0;
}
60 分值域很小：按值计数贪心测试点 10~12 ｜ O(V²)，V≤201

测试点 10~12 里 |ai| ≤ 100，值域只有 201 个整数。 把相同值的水晶合并计数，枚举目标值 V，代价 V−a ≥ 0， 显然优先用代价小的（也就是 a 大的），从 a=V 往下贪心取； 某一档取不完就说明预算耗尽，直接退出。整套做法与 n 无关。

// crystal 部分分 3：值域很小时按值计数（测试点 10~12：|a_i| <= 100, m <= 1e8）
// 枚举最终能量值 V（在值域内），价值 a 的水晶改成 V 的代价是 V-a >= 0，
// 显然优先用代价小的（也就是 a 大的），于是从 a=V 往下贪心取，直到 m 用完。
// O(V^2)，与 n 无关。
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    freopen("crystal.in", "r", stdin);
    freopen("crystal.out", "w", stdout);
    int n; ll m;
    scanf("%d %lld", &n, &m);
    const int LO = -105, HI = 105;
    vector<ll> c(HI - LO + 1, 0);
    for (int i = 1; i <= n; i++) { ll a; scanf("%lld", &a); c[a - LO]++; }
    int ans = 1;
    for (int V = LO; V <= HI; V++) {
        ll rest = m, got = 0;
        for (int a = V; a >= LO; a--) {
            ll cntv = c[a - LO];
            if (!cntv) continue;
            ll unit = V - a;
            if (unit == 0) { got += cntv; continue; }
            ll can = rest / unit;
            if (can >= cntv) { rest -= cntv * unit; got += cntv; }
            else { got += can; rest = 0; break; }
        }
        if (got > ans) ans = (int)got;
    }
    printf("%d\n", ans);
    return 0;
}
75 分特殊性质 A：ai = i，直接套公式测试点 13~15 ｜ O(log n)

因为 ai = i，序列天然递增。取末尾为 a[r] 的连续 k 个，代价是

k·r − (r−k+1 + … + r) = k(k−1)/2

和 r 无关——所以只要找最大的 k 使 k(k−1)/2 ≤ m，再与 n 取 min 即可。 不用排序、不用前缀和，一行二分搞定。注意 m ≤ 10¹⁸ 时 k 约为 1.5×10⁹， k(k−1)/2 必须用 __int128 算。

crystal_A.cpp公式 + 二分（性质 A）37 行▾ 
// crystal 部分分：特殊性质 A（a_i = i），O(1)
// 结论：排序后取连续 k 个（结尾为 r），代价 = k*r - (r-k+1 + ... + r) = k(k-1)/2，
//       与 r 无关！所以只要最大的 k 使 k(k-1)/2 <= m，再与 n 取 min。
// 期望得分：测试点 13~15（累计 75）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef __int128 lll;
int n; ll m;

// 最大的 k 使 k(k-1)/2 <= m（k 用 __int128 防溢出）
ll solve(ll m) {
    // k 的上界约 1.5e9（m <= 1e18 时），二分即可，不必推公式
    ll lo = 1, hi = 2000000000LL, best = 1;
    while (lo <= hi) {
        ll mid = lo + (hi - lo) / 2;
        lll cost = (lll)mid * (mid - 1) / 2;
        if (cost <= (lll)m) best = mid, lo = mid + 1;
        else hi = mid - 1;
    }
    return best;
}

int main() {
    freopen("crystal.in", "r", stdin);
    freopen("crystal.out", "w", stdout);
    scanf("%d %lld", &n, &m);
    for (int i = 1; i <= n; i++) {
        ll x;
        scanf("%lld", &x);  // 性质 A 下保证 x == i，直接丢弃
    }
    ll k = solve(m);
    if (k > n) k = n;       // 不能超过总数
    if (k < 1) k = 1;       // 单颗水晶代价 0，至少 1 颗
    printf("%lld\n", k);
    return 0;
}
90 分特殊性质 B：等差数列测试点 16~18 ｜ O(n log n)

设公差绝对值为 d（升序后相邻差就是它）。同样取连续 k 个，代价变成

d · k(k−1)/2 ≤ m

和 A 只差一个系数 d。要特判 d = 0：所有水晶一样，一颗都不用动，答案就是 n。 另外 d 本身可达 2×10¹²，乘上 k² 一定爆 long long，务必 __int128。

// crystal 部分分：特殊性质 B（a 是等差数列），O(log n)
// 设公差 d（可能为负，排序后按 |d| 算）。连续 k 个（结尾为 r）的代价
//   = k*a[r] - (a[r] + a[r-1] + ... + a[r-k+1]) = |d| * k(k-1)/2
// 与 r 无关，于是答案 = min(n, 最大的 k 使 |d|*k(k-1)/2 <= m)。
// 特判 d = 0：全部相同，直接输出 n。
// 期望得分：测试点 16~18（累计 90）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef __int128 lll;
int n; ll m;
ll a[500005];

int main() {
    freopen("crystal.in", "r", stdin);
    freopen("crystal.out", "w", stdout);
    scanf("%d %lld", &n, &m);
    for (int i = 1; i <= n; i++) scanf("%lld", &a[i]);

    if (n == 1) { printf("1\n"); return 0; }

    sort(a + 1, a + n + 1);               // 升序后相邻差就是 |d|
    ll d = a[2] - a[1];
    if (d == 0) {                         // 全是同一个值，一颗石头都不用动
        printf("%d\n", n);
        return 0;
    }

    int lo = 1, hi = n, best = 1;         // 至少 1 颗（代价 0）
    while (lo <= hi) {
        int mid = (lo + hi) >> 1;
        lll cost = (lll)d * mid * (mid - 1) / 2;
        if (cost <= (lll)m) best = mid, lo = mid + 1;
        else hi = mid - 1;
    }
    printf("%d\n", best);
    return 0;
}
100 分前缀和 + 二分（正解）测试点 19,20（也覆盖全部）｜ O(n log n)

排序、求前缀和。对每个右端点 r，在 k ∈ [1,r] 上二分最大的可行 k：

cost(r,k) = k·a[r] − (pre[r] − pre[r−k]) ≤ m

单调性证明：cost(r,k+1) − cost(r,k) = a[r] − a[r−k] ≥ 0（排序后），所以对 k 单调不降，可二分。 取所有 r 上的最大值即为答案。

溢出：k·a[r] 最大约 5×10⁵ × 10¹² = 5×10¹⁷，在 long long 内； 但前缀和可以为负，相减可能到 10¹⁸ 量级，用 __int128 最稳。

这一份代码同时能过性质 A、B 两档（cost 公式自动退化成 d·k(k−1)/2）， 所以正式比赛里其实可以只写这一份，前面几档是给你在考场上「先拿分再提速」的路线。

// crystal 正解：排序 + 前缀和 + 对每个“目标值”二分个数，O(n log n)
// 关键结论：若最终能量值定为 a[r]，则选 a[r] 前面（含自己）最大的 k 个最省，
//           代价 cost(r,k) = k*a[r] - (pre[r]-pre[r-k])，对 k 单调不降 -> 可二分
// 期望得分：100
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef __int128 lll;
int n; ll m;
int main() {
    freopen("crystal.in", "r", stdin);
    freopen("crystal.out", "w", stdout);
    scanf("%d %lld", &n, &m);
    vector<ll> a(n + 1), pre(n + 1, 0);
    for (int i = 1; i <= n; i++) scanf("%lld", &a[i]);
    sort(a.begin() + 1, a.end());
    for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
    int ans = 1;
    for (int r = 1; r <= n; r++) {
        int lo = 1, hi = r, best = 0;
        while (lo <= hi) {
            int k = (lo + hi) >> 1;
            lll cost = (lll)k * a[r] - (lll)(pre[r] - pre[r - k]);
            if (cost <= (lll)m) best = k, lo = k + 1;
            else hi = k - 1;
        }
        ans = max(ans, best);
    }
    printf("%d\n", ans);
    return 0;
}

  • 答案下界是 1（单颗水晶代价 0），不是 0。
  • ai 可以是负数，前缀和会出现负数，不要把代价先取绝对值。
  • 别用「取最小的 k 个」那一版（见上面的反例）。

T2地铁 subway

时限 1.0s内存 1024MiB25 个测试点，每点 4 分目录 subway

建模：把「线路」当成点

题目只统计步行时间，坐车时间不计。于是：

  1. 同一条线路上换站是 0 代价，所以一条线路上的所有站台可以缩成一个点。
  2. 站 v 的站厅与经过它的线路 i 之间连一条权为 t[i][v] 的边；进站、出站各付一次 w。
  3. 若线路 i、j 都经过站 v，在 v 换乘的代价是 t[i][v] + t[j][v]（站台→站厅→站台，不经过地铁口，所以不加 wv）。

由此得到只含 n 个点的「线路图」：边权 W[i][j] = minv 同时属于 i,j(t[i][v] + t[j][v])， 再令 D[i][j] 为线路图上的最短路，则询问答案为

ans(x,y) = wx + wy + mini∋x, j∋y ( t[i][x] + D[i][j] + t[j][y] )

为什么可以这么压？任何一条真实路线都是「进站 → 上首条线 → 若干次换乘 → 下末条线 → 出站」， 每次换乘恰好对应线路图上一条边；反过来线路图上每条边都对应一个真实的换乘站， 且上车后可以在该线内免费坐到下一个换乘站。两边一一对应，最优值不变。

规模的关键：n ≤ 500。所以 Floyd 是 O(n³) = 1.25×10⁸，可以放心写； 而「每个站经过的线路数 ≤ d ≤ 20」保证了每站最多 C(20,2)=190 对线路要建边， 又因为 Σk ≤ 5×10⁵，总对数不会超过 5×10⁵/20 × 190 ≈ 4.75×10⁶。

写的时候三个细节

  • 换乘不加 w。题面特别强调「站内换乘只需站台→站厅→站台」，w 只算进出站各一次。
  • 同一条线的两个人不用换乘：D[i][i]=0 自动覆盖「x、y 在同一条线上」的情况。
  • 答案用 long long：单次换乘可达 2×10⁹，路径上多次相加会超过 int。

时限 1.0s内存 1024MiB25 个测试点，每点 4 分多组数据，Σn ≤ 2×10⁶

先把图想清楚

岛屿按坐标排好，边是 |xi−xj| ≤ d。三个基本事实：

  1. 最短路只会向一个方向飞。若某条最短路里出现「先向左再向右」，把这一绕路直接删掉不会变长， 所以统计时只需考虑下标单调递增的路径（这一点用 BFS 暴力对拍可以验证）。
  2. 一步能到的最右岛 R[i]（双指针 O(n)）决定了一切： 从 u 出发，飞 t 步能到的正好是连续区间 (Pt−1, Pt]， 其中 P0=u, Pt=R[Pt−1]。也就是说第 t 层就是一段连续下标。
  3. 记 f(v) = min{w : R[w] ≥ v}（能一步到 v 的最左岛，双指针预处理）， 那么计算 v 的最短路条数时，它在本层的前驱就是区间 [max(f(v), Pt−2+1), Pt−1]， 条数等于上一层这段区间的前缀和之差。

由此得到第一个有用的性质：同一层里 v 越大，f(v) 越大，最短路条数 cnt(v) 单调不增。 于是「下一层每个点的条数」只由当前层最右的两个点决定，这就是状态压缩的全部由来。

两个计数式子（正解的核心）

设当前层右端为 p、上一层右端为 q，下一层右端 t = R[p]。下一层的点分三类：

  • one = t − max(p, R[p−1]) 个点只能看见 p（f(v)=p），它们的条数 = cnt(p)；
  • two = max(0, R[p−1] − max(p, R[p−2])) 个点恰好能看见 p−1 和 p， 条数 = cnt(p−1) + cnt(p)；
  • 剩下的点前驱至少三个，条数 ≥ 3，与答案无关。

所以「恰好两条」的终点只可能出现在前两类里，于是每个起点只需要记三种状态：

状态含义本层贡献下一层去向
Acnt(p)=1 且 cnt(p−1)=1（p−1 在本层内） two 个点（两条唯一路径在 v 处汇合）见下表
Bcnt(p)=1，但 cnt(p−1)≥2 或 p−1 不在本层0见下表
Ccnt(p)=2one 个点（两条路径都经过 p 继续往上）见下表
丢弃cnt(p)≥3条数只会越来越大，此后永远不再贡献，直接停

状态转移（one≥2 表示下一层最右两点都只看见 p，one=1 表示最右看见 p、次右看见 p−1 及更左）：

条件A / B（cnt(p)=1）C（cnt(p)=2）
one ≥ 2新层两点条数都是 1 → A新层两点条数都是 2 → C
one = 1次右点 ≥2 → BC
one = 0, two > 0只有 A 能续：条数 1+1=2 → C；B 直接丢丢（2+2≥3）
one = 0, two = 0丢丢

最后一步是本题的“点睛”：转移只跟 p（层右端）和状态有关，跟起点是谁无关。 于是把所有起点按「当前层右端 p」聚合起来：A[p], B[p], C[p] 记有多少个起点处于该状态， 从 p=1 扫到 n，一边算答案一边把状态搬到 t=R[p]。整题 O(n) 一趟扫完。

48 分特殊性质 A：坐标等差测试点 11,12 ｜ O(n)

设公差 s，一步最多跨 m = ⌊d/s⌋ 段。从 u 到 v（下标差 Δ） 要走 k = ⌈Δ/m⌉ 步，最短路径条数 = 把 Δ 拆成 k 个 [1,m] 整数之和的方案数。

  • k=1：1 条；
  • k=2：方案数 2m−Δ+1，等于 2 ⇔ Δ = 2m−1（需 m≥2）；
  • k≥3：方案数已经 ≥ 3。

所以答案就是满足 Δ = 2m−1 的岛对数，即 max(0, n−(2m−1))。 用样例验证：x=1..4, d=2 → m=2, 2m−1=3，答案 4−3=1 ✓； x=1..5, d=2 → 2 ✓。

56 分特殊性质 B：存在枢纽岛测试点 13,14 ｜ O(n)

存在一座岛 h 能直飞所有岛，于是任意两岛之间最多 2 步： 能一步到就 1 条路径；否则（xv−xu > d）恰好 2 步，条数 = 能同时被 u 与 v 一步到达的中间岛个数。

中间岛下标落在 [L(v), R(u)]（L(v)= 能一步到 v 的最左岛，R(u)= u 一步能到的最右岛）， 个数是 R(u)−L(v)+1。要它等于 2 ⇔ R(u) = L(v)+1。 用 cntR[z] = #{u : R(u)=z} 做一次桶统计即可 O(n)。

            if (k <= n) L[v] = k; else L[v] = n + 1;

T4研发基地 hub

时限 3.0s内存 1024MiB20 个测试点，每点 5 分多组数据，Σn ≤ 2×10⁶

第一步：二分答案

「最小化最大通勤距离」是典型的二分：二分半径 R，判定 「能否选恰好 k 个点、形成不超过两个连通块，使全树每个点都在某个选中点的 R 邻域里」。 R 为 0..n，判定 O(n)，总 O(n log n)。

第二步：一个连通块时的最小点数 = R-核心

对一个连通块，最少要多少点？答案是逐轮剥叶后的核心：

  1. 按「每轮把所有叶子（度为 1）剥掉」给每个点标上 lev[v]（第几轮被剥掉，最后剩下的点标最大轮）；
  2. 核心 = { v : lev[v] > R }。

两个方向都要说清楚：

  • 核心够用：若 lev[x] = ℓ ≤ R，则 x 到核心的距离恰好是 ℓ，所以核心能 R-覆盖全树。 这正是「lev 必须逐轮剥叶算，不能用 DFS 深度代替」的原因。
  • 核心不能少：核心上的点两侧都有长度 > R 的深枝；若它不在选中集合里，两侧就都得选点，而连通性又强迫它自己被选。

所以一个连通块的最小点数就是 C = |核心|；C ≤ k 时直接可行（C=0 时放 1 个点即可，因为 k≥2）。 多出来的名额可以随便往连通块里加相邻点，所以「恰好 k 个」不构成额外约束。

第三步：两个连通块 = 在核心的「度-2 链」上删一段

核心是一棵树。用两个连通块，相当于把核心挖掉一条连续路径让它裂成两块： 挖掉的点必须是核心中度数为 2 的链上的连续一段（度数 ≥3 的点挖掉会裂成三块，核心叶子则永远挖不动）。 于是只要枚举「在哪条链上、挖哪一段」。

设链上第 i 个点被挖掉，左右两侧保留的点分别是第 L、Rgt 个， ai = 挂在它身上的、已被剥落的枝条最大深度（即不在核心里的邻点的最大 lev）， 那么枝条最深处到最近保留点的距离必须 ≤ R：

min(i−L, Rgt−i) + ai ≤ R

把「左覆盖」和「右覆盖」拆开：记 posti = i+ai−R、hi = i+R−ai， 可以证明上式 ⇔「不存在 i 同时满足 posti > L 与 hi < Rgt」。 于是对每个 L，能取到的最右端是

Rgtmax(L) = min(m+1, min{ hi : posti > L })

把 h 按 post 做一次后缀最小值，就得到所有 L 的答案，链上线性扫完。 链的处理是 O(n)，整体判定 O(n)，总 O(n log n)（实测 10⁶ 点约 0.17s）。

// hub 暴力：枚举所有大小恰为 k 的点集，判断连通块数 <= 2 并计算通勤半径
// 仅用于对拍（n <= 12）
#include <bits/stdc++.h>
using namespace std;
int n, k;
vector<int> adj[13];
int main() {
    freopen("hub.in", "r", stdin);
    freopen("hub.out", "w", stdout);
    int c, t;
    if (scanf("%d %d", &c, &t) != 2) return 0;
    while (t--) {
        scanf("%d %d", &n, &k);
        for (int i = 0; i <= n; i++) adj[i].clear();
        for (int i = 1; i < n; i++) {
            int u, v; scanf("%d %d", &u, &v);
            adj[u].push_back(v); adj[v].push_back(u);
        }
        int best = 1e9;
        for (int mask = 0; mask < (1 << n); mask++) {
            if (__builtin_popcount(mask) != k) continue;
            // 连通块数
            vector<int> comp(n + 1, -1);
            int cc = 0;
            for (int s = 1; s <= n; s++) if ((mask >> (s - 1) & 1) && comp[s] < 0) {
                queue<int> q; q.push(s); comp[s] = cc;
                while (!q.empty()) {
                    int u = q.front(); q.pop();
                    for (int v : adj[u]) if ((mask >> (v - 1) & 1) && comp[v] < 0) comp[v] = cc, q.push(v);
                }
                cc++;
            }
            if (cc > 2) continue;
            // 多源 BFS 求半径
            vector<int> d(n + 1, -1);
            queue<int> q;
            for (int i = 1; i <= n; i++) if (mask >> (i - 1) & 1) d[i] = 0, q.push(i);
            int r = 0;
            while (!q.empty()) {
                int u = q.front(); q.pop();
                r = max(r, d[u]);
                for (int v : adj[u]) if (d[v] < 0) d[v] = d[u] + 1, q.push(v);
            }
            best = min(best, r);
        }
        printf("%d\n", best);
    }
    return 0;
}
35 分二分 + 枚举断边 + 两侧剥叶测试点 4~10 ｜ O(n² log n)

思路最直观的一档：既然只有两个连通块，那一定是某条边被断开后的两侧。 于是二分 R，判定时枚举每条边：两侧各自做一次逐轮剥叶， 各取「轮数 > R」的点数相加，与 k 比较。

一个必须注意的细节：某侧剥完一棵都不剩时，仍要放 1 个点（返回 max(1, 核心大小)）， 否则会低估点数、把不可行的 R 判成可行。我第一次写就漏了这个 max(1,…)，对拍立刻挂。

// hub 部分分：二分答案 + 枚举“断哪条边” + 两侧分别逐轮剥叶
// 结论：一个连通块时的最小点数 = 逐轮剥叶后“轮数 > R”的点数（R-核心）；
//       两个连通块 = 把某条边断开，两侧各取自己的 R-核心，点数相加。
// O(n^2 log n)，可过 n<=4000（测试点 4~10，约 35 分）
#include <bits/stdc++.h>
using namespace std;
static int n, k;
static vector<int> head, eto, enxt;
static vector<int> mark, lev, deg;
static int tag = 0;
static vector<int> nodes, bfs, q;

// 对 nodes（当前 tag 标记的点集，必须连通）做逐轮剥叶，返回“轮数 > R”的点数
static int coreSize(int R) {
    for (int i = 0; i < (int)nodes.size(); i++) { int u = nodes[i]; deg[u] = 0; lev[u] = 0; }
    for (int i = 0; i < (int)nodes.size(); i++) {
        int u = nodes[i];
        for (int e = head[u]; e != -1; e = enxt[e]) if (mark[eto[e]] == tag) deg[u]++;
    }
    q.clear();
    for (int i = 0; i < (int)nodes.size(); i++) {
        int u = nodes[i];
        if (deg[u] <= 1) { lev[u] = 1; q.push_back(u); }
    }
    for (int h = 0; h < (int)q.size(); h++) {
        int u = q[h];
        for (int e = head[u]; e != -1; e = enxt[e]) {
            int v = eto[e];
            if (mark[v] != tag || lev[v]) continue;
            if (--deg[v] <= 1) { lev[v] = lev[u] + 1; q.push_back(v); }
        }
    }
    int c = 0;
    for (int i = 0; i < (int)nodes.size(); i++) if (lev[nodes[i]] > R) c++;
    // 核心为空时仍至少要放一座基地
    return c > 0 ? c : 1;
}

// 从 s 出发、不跨越边 (s0,t0) 收集一侧连通块，返回点集大小化在 nodes 里
static void collect(int s, int banned) {
    tag++;
    nodes.clear(); bfs.clear();
    bfs.push_back(s); mark[s] = tag;
    for (int i = 0; i < (int)bfs.size(); i++) {
        int x = bfs[i];
        nodes.push_back(x);
        for (int f = head[x]; f != -1; f = enxt[f]) {
            int y = eto[f];
            if (y == banned || mark[y] == tag) continue;
            mark[y] = tag; bfs.push_back(y);
        }
    }
}

int main() {
    freopen("hub.in", "r", stdin);
    freopen("hub.out", "w", stdout);
    int c, t;
    if (scanf("%d %d", &c, &t) != 2) return 0;
    while (t--) {
        scanf("%d %d", &n, &k);
        head.assign(n + 1, -1);
        eto.assign(2 * (n + 1), 0); enxt.assign(2 * (n + 1), 0);
        int ec = 0;
        vector<pair<int, int> > es;
        for (int i = 1; i < n; i++) {
            int u, v; scanf("%d %d", &u, &v);
            eto[ec] = v; enxt[ec] = head[u]; head[u] = ec++;
            eto[ec] = u; enxt[ec] = head[v]; head[v] = ec++;
            es.push_back(make_pair(u, v));
        }
        mark.assign(n + 1, 0); lev.assign(n + 1, 0); deg.assign(n + 1, 0);
        nodes.reserve(n); bfs.reserve(n); q.reserve(n);
        auto feasible = [&](int R) -> bool {
            collect(1, 0);                       // 情况一：一个连通块
            if (coreSize(R) <= k) return true;
            for (auto &e : es) {                 // 情况二：断开一条边
                int a = e.first, b = e.second;
                collect(a, b);
                int tot = coreSize(R);
                if (tot > k) continue;
                collect(b, a);
                tot += coreSize(R);
                if (tot <= k) return true;
            }
            return false;
        };
        int lo = 0, hi = n, ans = n;
        while (lo <= hi) {
            int mid = (lo + hi) >> 1;
            if (feasible(mid)) { ans = mid; hi = mid - 1; }
            else lo = mid + 1;
        }
        printf("%d\n", ans);
    }
    return 0;
}
100 分核心 + 在度-2 链上删一段（正解）测试点 18~20（覆盖全部）｜ O(n log n)

判定函数分四步：

  1. 整棵树剥叶一次（O(n)，只做一次，与 R 无关），得到 lev[]；
  2. 核心 = {lev > R}，大小 C；C ≤ k 直接可行；
  3. 算核心内部度数，把「核心内度数为 2」的点串成若干条链；
  4. 对每条链：算 ai（悬挂枝条最大 lev）→ posti, hi → 按 post 求 h 的后缀最小值 → 对每个 L 取 Rgtmax， 得到这条链最多能删几个点。所有链取最大，判 C − 最大删除数 ≤ k。

边界：链的两端外各有一个「外部邻居」（度数 ≠ 2 的核心点），删段可以贴到链头/链尾， 所以 L 从 0 开始、Rgt 最大到 m+1； ai 只统计不在核心的邻点。

整题复杂度 O(n log n)：二分 O(log n) 轮，每轮 O(n)。 实现上全程用"边表 + 数组 + 队列"，不递归，1e6 点也不会爆栈。

hub_ac.cpp核心 + 链上删段二分答案（正解）114 行▾ 
// hub 正解：二分答案 + 逐轮剥叶（R-核心）+ 在核心的“度数为 2 的链”上删一段
// 1) 二分通勤半径 R
// 2) 逐轮剥叶求 lev[v]（第几轮被剥掉）；lev[v] > R 的点构成 R-核心。
//    核心恰好是“一个连通块”时的最小方案（核心本身就能 R-覆盖全树）
// 3) 若 |核心| <= k 直接可行；否则必须用两个连通块 = 在核心的一条度-2 链上删掉连续一段。
//    被删的点 i（链上第 i 个，相邻保留点是第 L 和第 Rgt 个）必须满足
//        min(i-L, Rgt-i) + a_i <= R      （a_i = 它悬挂的已剥落枝条的最大深度）
//    等价于：不存在 i 使 post_i > L 且 h_i < Rgt，其中 post_i = i+a_i-R, h_i = i+R-a_i。
//    于是对每个 L 取 Rgt <= min(m+1, min{h_i : post_i > L})，线性扫一遍取最大可删长度。
// 期望得分：100   复杂度 O(n log n)
#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9;
static int n, k;
static int head[1000005], eto[2000005], enxt[2000005], ecnt;
static int deg[1000005], lev[1000005], que[1000005];
static int cdeg[1000005];
static char inCore[1000005], used[1000005];
static int chainBuf[1000005], minAt[1000005], smin[1000005];

static inline void addEdge(int u, int v) { eto[ecnt] = v; enxt[ecnt] = head[u]; head[u] = ecnt++; }

// 逐轮剥叶：lev[v] = 第几轮被剥掉
static void peel() {
    for (int i = 1; i <= n; i++) { deg[i] = 0; lev[i] = 0; }
    for (int u = 1; u <= n; u++)
        for (int e = head[u]; e != -1; e = enxt[e]) deg[u]++;
    int h = 0, t = 0;
    for (int i = 1; i <= n; i++) if (deg[i] <= 1) { lev[i] = 1; que[t++] = i; }
    while (h < t) {
        int u = que[h++];
        for (int e = head[u]; e != -1; e = enxt[e]) {
            int v = eto[e];
            if (lev[v]) continue;
            if (--deg[v] <= 1) { lev[v] = lev[u] + 1; que[t++] = v; }
        }
    }
}

static bool check(int R) {
    int C = 0;
    for (int i = 1; i <= n; i++) { inCore[i] = (lev[i] > R); C += inCore[i]; }
    if (C <= k) return true;               // 一个连通块就够（含核心为空、k>=2）
    for (int i = 1; i <= n; i++) cdeg[i] = 0;
    for (int u = 1; u <= n; u++) if (inCore[u])
        for (int e = head[u]; e != -1; e = enxt[e]) if (inCore[eto[e]]) cdeg[u]++;
    for (int i = 1; i <= n; i++) used[i] = 0;
    int bestDel = 0;
    for (int s = 1; s <= n; s++) {
        if (!inCore[s] || cdeg[s] == 2) continue;
        for (int e = head[s]; e != -1; e = enxt[e]) {
            int v = eto[e];
            if (!inCore[v] || cdeg[v] != 2 || used[v]) continue;
            // 拉出一条度-2 链，第 0 个位置是 s，第 1..m 是链上点
            int m = 0, cur = v, prev = s;
            while (true) {
                chainBuf[++m] = cur; used[cur] = 1;
                int nx = -1;
                for (int f = head[cur]; f != -1; f = enxt[f]) {
                    int w = eto[f];
                    if (w != prev && inCore[w] && cdeg[w] == 2) { nx = w; break; }
                }
                if (nx == -1) break;
                prev = cur; cur = nx;
            }
            for (int i = 0; i <= m + 2; i++) minAt[i] = INF;
            for (int i = 1; i <= m; i++) {
                int u = chainBuf[i], a = 0;
                for (int f = head[u]; f != -1; f = enxt[f]) {
                    int w = eto[f];
                    if (!inCore[w] && lev[w] > a) a = lev[w];
                }
                int post = i + a - R; if (post < 0) post = 0; if (post > m) post = m;
                int hv = i + R - a;           // h_i，超过 m+1 就没意义
                if (hv > minAt[post]) continue;
                minAt[post] = hv;
            }
            smin[m + 2] = INF;
            for (int i = m + 1; i >= 0; i--) smin[i] = min(minAt[i], smin[i + 1]);
            for (int L = 0; L <= m - 1; L++) {
                int maxRgt = min(m + 1, smin[L + 1]);   // 右端最多到 m+1
                if (maxRgt > L + 1) bestDel = max(bestDel, maxRgt - L - 1);
            }
        }
    }
    return C - bestDel <= k;
}

int main() {
    freopen("hub.in", "r", stdin);
    freopen("hub.out", "w", stdout);
    int c, t;
    if (scanf("%d %d", &c, &t) != 2) return 0;
    while (t--) {
        scanf("%d %d", &n, &k);
        ecnt = 0;
        for (int i = 1; i <= n; i++) head[i] = -1;
        for (int i = 1; i < n; i++) {
            int u, v; scanf("%d %d", &u, &v);
            addEdge(u, v); addEdge(v, u);
        }
        peel();
        int mx = 0;
        for (int i = 1; i <= n; i++) mx = max(mx, lev[i]);
        int lo = 0, hi = mx, ans = mx;
        while (lo <= hi) {
            int mid = (lo + hi) >> 1;
            if (check(mid)) { ans = mid; hi = mid - 1; }
            else lo = mid + 1;
        }
        printf("%d\n", ans);
    }
    return 0;
}

易错点

  • lev 必须是「逐轮剥叶」的层数，用 BFS/DFS 深度会算错核心。
  • 最后剩下的那个点（剥到度为 0）也要给它赋 lev，否则它不会被划进核心。
  • 核心为空时要放 1 个点，别返回 0。
  • 多组数据、Σn=2×10⁶：只清空到 n，且不要用 memset 清全数组。
