T1 · 廊桥分配

题意

机场有 n 个廊桥，要分给国内区和国际区（国内航班只能停国内区）。给定 m1 架国内航班和 m2 架国际航班的 (抵达时刻, 离开时刻)，飞机抵达时若对应区域还有空桥就停靠，否则去远机位（先到先得）。问怎么分配廊桥，能让停靠廊桥的飞机总数最多。

数据范围与子任务

洛谷题面只给了百分比，按"20 个测试点各 5 分"换算：

测试点 分值 累计 n ≤ m1+m2 ≤
1 ~ 4 20 20 100 100
5 ~ 8 20 40 5000 5000
9 ~ 20 60 100 105 105

所有时刻是 ≤ 108 的互不相同的正整数，每架飞机 a < b。

先把问题拆开

设国内分到 x 个廊桥，国际分到 n−x 个。两个区互不影响（各自的航班只能用自己区的桥），所以

答案 = maxx=0..n { f(x) + g(n−x) }

其中 f(x) = 国内 x 个桥能停几架，g(y) = 国际 y 个桥能停几架。

所以整道题就变成一件事：对每个 x，快速算出 f(x)。

还有一个显然的性质：f(x) 关于 x 单调不减（桥多了不可能停得更少），且 f(x) ≤ m1。

第一档：20 分（测试点 1~4，n ≤ 100）

最直白的写法：枚举 x，然后老老实实模拟 x 个廊桥。

模拟的关键只有一句：同一区域内，具体停在哪个廊桥不影响后续。因为廊桥是对称的，一架飞机只要"有空桥"就能停，选择哪个空桥不会改变"还有几个空桥"这个事实。所以模拟时只需要维护"每个廊桥下次空闲的时刻"，遇到飞机就从头找一个空闲的。

复杂度 O(n · m · n) ≈ 100 × 100 × 100 = 106。

C++
// CSP-S 2021 T1 廊桥分配 —— 20 分档（n,m1+m2 <= 100）
// 做法：枚举分给国内区的廊桥数 x = 0..n，两边分别 O(m*min(x,m)) 暴力模拟。
// 复杂度 O(n * m * n)，在 n,m <= 100 时约 1e6 次操作。
#include <bits/stdc++.h>
using namespace std;

int n, m1, m2;
struct Fl { int a, b; };           // a: 抵达时刻, b: 离开时刻
vector<Fl> u, v;                   // 国内 / 国际航班

bool cmpA(const Fl &x, const Fl &y) { return x.a < y.a; }

// t 已按抵达时刻升序。k 个廊桥时能停靠的航班数。
int calc(const vector<Fl> &t, int k) {
    if (k == 0) return 0;
    vector<int> fre(k, 0);         // fre[i] = 第 i 个廊桥下次空闲的时刻（0 表示一直空着）
    int c = 0;
    for (const auto &f : t) {
        for (int i = 0; i < k; i++) {
            if (fre[i] < f.a) {    // 时刻互不相同，等价于 <=
                fre[i] = f.b;      // 占用到 b 时刻
                c++;
                break;             // 有桥就停，"先到先得"，编号不影响结果
            }
        }
    }
    return c;
}

int main() {
    if (scanf("%d %d %d", &n, &m1, &m2) != 3) return 0;
    u.resize(m1); v.resize(m2);
    for (auto &x : u) scanf("%d %d", &x.a, &x.b);
    for (auto &x : v) scanf("%d %d", &x.a, &x.b);
    sort(u.begin(), u.end(), cmpA);
    sort(v.begin(), v.end(), cmpA);

    int ans = 0;
    for (int x = 0; x <= n; x++) {
        int cur = calc(u, x) + calc(v, n - x);
        if (cur > ans) ans = cur;
    }
    printf("%d\n", ans);
    return 0;
}

第二档：40 分（测试点 5~8，n,m ≤ 5000）

上面那份代码慢在"每次找空桥要扫 x 个廊桥"。其实只要维护一个小根堆，存"正在占用廊桥的飞机的离开时刻"：

  • 每来一架飞机，先把堆里所有"离开时刻 < 抵达时刻"的弹掉（这些廊桥空出来了）；
  • 如果堆的大小 < k，说明还有空桥，占用一个（把 b 压进去），计数 +1；
  • 否则飞远机位。

复杂度 O(n · m log m)。实测 n = m = 5000 时 0.27s，能过。

C++
// CSP-S 2021 T1 廊桥分配 —— 40 分档（n,m1+m2 <= 5000）
// 做法：仍然枚举 x，但单次模拟从 O(m*n) 降到 O(m log m)：
//   用一个小根堆存"正在占用廊桥的飞机离开时刻"，离开时刻 < 当前抵达时刻的弹出，
//   堆里剩下的个数就是"被占用的廊桥数"，它小于 k 就说明还有空桥。
// 复杂度 O(n * m log m)。
#include <bits/stdc++.h>
using namespace std;

int n, m1, m2;
struct Fl { int a, b; };
vector<Fl> u, v;

bool cmpA(const Fl &x, const Fl &y) { return x.a < y.a; }

int calc(const vector<Fl> &t, int k) {
    if (k == 0) return 0;
    priority_queue<int, vector<int>, greater<int> > q;   // 占用中的离开时刻
    int c = 0;
    for (const auto &f : t) {
        while (!q.empty() && q.top() < f.a) q.pop();     // 已经飞走的释放
        if ((int)q.size() < k) { q.push(f.b); c++; }     // 还有空桥
    }
    return c;
}

int main() {
    if (scanf("%d %d %d", &n, &m1, &m2) != 3) return 0;
    u.resize(m1); v.resize(m2);
    for (auto &x : u) scanf("%d %d", &x.a, &x.b);
    for (auto &x : v) scanf("%d %d", &x.a, &x.b);
    sort(u.begin(), u.end(), cmpA);
    sort(v.begin(), v.end(), cmpA);

    int ans = 0;
    for (int x = 0; x <= n; x++) {
        int cur = calc(u, x) + calc(v, n - x);
        if (cur > ans) ans = cur;
    }
    printf("%d\n", ans);
    return 0;
}

第三档：100 分（n, m ≤ 105）

40 分做法要枚举 n 个 x，每个 x 重跑一遍模拟，总复杂度带着一个 n。想要 100 分，必须一次模拟求出所有 f(k)。

核心引理（虚拟廊桥）

先假装廊桥无限多，按抵达时刻依次处理每架飞机：

  • 若有空闲廊桥，就停在编号最小的那个；
  • 若没有（1..mx 号全忙），就新开一个编号 mx+1 的廊桥。

记 cnt[k] = 编号为 k 的廊桥一共服务了多少架飞机。

引理：只给 k 个廊桥时能停的飞机数 f(k) = cnt[1] + cnt[2] + ⋯ + cnt[k]。

证明：对航班序列做归纳。假设在处理完前若干架飞机后，"无限桥模拟中编号 ≤ k 的桥的占用状态"与"只有 k 个桥的模拟状态"完全一致，且两者对每架飞机的决策也一致：

  • 若无限桥模拟中被分到编号 j ≤ k 的桥，说明当时 1..k 中有空桥，k 桥模拟同样会停靠，且选的是同一个编号（都取最小编号）；
  • 若被分到编号 j > k，说明当时 1..k 全被占用，k 桥模拟只能让这架飞机飞远机位。

两种模拟的"归还顺序"也一致（都按离开时刻弹出），所以归纳成立。于是"编号 ≤ k 的桥服务过的航班"恰好就是"k 个桥能停的航班"，取前缀和即得。□

实现

两个堆：

  • occ：小根堆，元素是 (离开时刻, 廊桥编号)，按离开时刻排序；
  • idle：小根堆，存空闲的廊桥编号。

每来一架飞机：先把 occ 里离开时刻 < 抵达时刻的弹出，把它们的编号丢进 idle；然后取 idle 的最小值（没有就新开编号）。最后 cnt[编号]++。

因为 idle 取的是最小编号，所以编号总是"连续"的：新开的编号一定是当前最大值 +1，cnt 数组可以边用边 push_back。

最后对 cnt 做前缀和，得到 f(0), f(1), …, f(mx)。注意当 x > mx 时 f 已经饱和（所有飞机都能停），取 f.back() 即可。

复杂度 O((m1+m2) log(m1+m2))。实测 n = m = 105 时 0.02s。

C++
// CSP-S 2021 T1 廊桥分配 —— 100 分（n, m1+m2 <= 1e5）
// 核心：把"n 个廊桥怎么分"变成"第 k 个廊桥服务了几架飞机"。
//   先假装廊桥无限多，给每架飞机分配一个编号，编号 = 它用的是"第几个"廊桥。
//   记 cnt[k] 为编号 k 的廊桥服务的飞机数，则"恰好给 k 个廊桥"时能停的架数
//   f[k] = cnt[1] + cnt[2] + ... + cnt[k]（前缀和）。
// 答案 = max_{x=0..n} f1[x] + f2[n-x]。
// 复杂度 O((m1+m2) log(m1+m2))。
#include <bits/stdc++.h>
using namespace std;

int n, m1, m2;
struct Fl { int a, b; };
vector<Fl> u, v;

bool cmpA(const Fl &x, const Fl &y) { return x.a < y.a; }

// 返回 s，s[k] = "给 k 个廊桥时能停靠的飞机数"，k = 0 .. 最大用到的编号
vector<long long> solve(const vector<Fl> &t) {
    // occ: (离开时刻, 廊桥编号)，按离开时刻小根堆；idle: 空闲廊桥编号小根堆
    priority_queue<pair<int, int>, vector<pair<int, int> >, greater<pair<int, int> > > occ;
    priority_queue<int, vector<int>, greater<int> > idle;
    vector<long long> cnt(1, 0);        // cnt[k]，1-indexed，cnt[0] 占位
    int mx = 0;
    for (const auto &f : t) {
        while (!occ.empty() && occ.top().first < f.a) {
            idle.push(occ.top().second); // 飞机飞走，廊桥回收
            occ.pop();
        }
        int id;
        if (!idle.empty()) { id = idle.top(); idle.pop(); }
        else { id = ++mx; cnt.push_back(0); }   // 没有空桥 -> 虚拟开一个新编号
        cnt[id]++;
        occ.push(make_pair(f.b, id));
    }
    for (size_t i = 1; i < cnt.size(); i++) cnt[i] += cnt[i - 1];
    return cnt;                          // cnt[k] = f(k)
}

int main() {
    if (scanf("%d %d %d", &n, &m1, &m2) != 3) return 0;
    u.resize(m1); v.resize(m2);
    for (auto &x : u) scanf("%d %d", &x.a, &x.b);
    for (auto &x : v) scanf("%d %d", &x.a, &x.b);
    sort(u.begin(), u.end(), cmpA);
    sort(v.begin(), v.end(), cmpA);

    vector<long long> f1 = solve(u), f2 = solve(v);
    // 取第 k 个值；k 超过数组范围说明廊桥多到能把所有飞机都停下，饱和
    auto get = [](const vector<long long> &s, int k) -> long long {
        if (k >= (int)s.size()) return s.back();
        return s[k];
    };

    long long ans = 0;
    for (int x = 0; x <= n; x++) {
        long long cur = get(f1, x) + get(f2, n - x);
        if (cur > ans) ans = cur;
    }
    printf("%lld\n", ans);
    return 0;
}

易错点

  1. x 可以取 0 和 n，循环要写 x = 0 .. n。
  2. n 可能远大于 m1（比如 n = 105、m1 = 1）。此时 f1(x) 早就饱和了，取下标前必须判越界，返回最后一个值。
  3. cnt 数组是边算边增长的，不要一开始开成 n+1 就以为够了 —— 编号最大只到"实际用到的桥数"，可能比 n 小得多，也可能（在 40 分写法里）需要判断 x > mx。
  4. 判"桥空了"用 离开时刻 < 抵达时刻。题目保证所有时刻互不相同，所以 < 和 ≤ 等价；但如果你自己造数据时出现了相等，语义上应该是"离开时刻 ≤ 抵达时刻"才算空。
  5. 答案最大是 m1+m2 ≤ 105，int 够；但用 long long 不会有任何代价。

T2 · 括号序列

题意

给一个长度 n 的串，只含 (、)、*、?。? 可以替换成三种字符之一。定义一个"符合规范的超级括号序列"：

  1. () 和 (S) 合法，其中 S 是非空且长度 ≤ k 的星串；
  2. 若 A、B 合法，则 AB、ASB 合法；
  3. 若 A 合法，则 (A)、(SA)、(AS) 合法；
  4. 只有这些。

问有多少种替换方式能让串合法，对 109+7 取模。

数据范围与子任务

测试点 分值 累计 n ≤ 特殊性质
1 ~ 3 15 15 15 无
4 ~ 8 25 40 40 无
9 ~ 13 25 65 100 无
14 ~ 15 10 75 500 S 串中仅含有 ?
16 ~ 20 25 100 500 无

1 ≤ k ≤ n ≤ 500。

一个必须先想明白的事实

任何合法串都以 ( 开头、以 ) 结尾。

证明：对规则归纳。规则 1、3 产生的串显然满足；规则 2 产生的 AB、ASB 以 A 的首字符开头（A 以 ( 开头）、以 B 的尾字符结尾（B 以 ) 结尾）。归纳成立。□

这条事实极其重要，它排除了"以 * 结尾"这种看起来像合法串、其实不是的东西 —— 我第一次写这题就是在这里多算了 4 种方案，样例 1 算出 9 而不是 5。

唯一块分解

既然合法串以 ( 开头，它一定能唯一地切成若干"块"：

文本
合法串 = B₁ B₂ … B_t     (t ≥ 1)
每块 B = '(' X ')' 或者 '(' X ')' S

其中：

  • S 是非空星串（长度 ≤ k）；
  • X ∈ { 空, S, A, SA, AS }，A 是合法串。

唯一性：第一块的位置由"第一个 ( 的匹配 )"唯一确定；紧跟的星串由"从这里开始连续多少个 *"唯一确定（且不能超 k）。所以分解无重复。

四个数组

设区间 [i,j]（1-indexed）：

数组 含义
Sg[i][j] [i,j] 本身是一个合法星串（非空、长度 ≤ k、每个位置都能是 *）
L[i][j] [i,j] 恰好是"一块括号" ( X )
N[i][j] [i,j] 是 (X) 紧接一段非空星串
E[i][j] [i,j] 是一个合法串

转移：

文本
L[i][j] = [s[i] 能是 '('] · [s[j] 能是 ')'] ·
          ( [j == i+1]                        // ()
          + Sg[i+1][j-1]                      // (S)
          + E[i+1][j-1]                       // (A)
          + Σ_m Sg[i+1][m] · E[m+1][j-1]      // (S A)
          + Σ_m E[i+1][m] · Sg[m+1][j-1] )    // (A S)

N[i][j] = Σ_m L[i][m] · Sg[m+1][j]            // (X) + 星串

E[i][j] = L[i][j] + Σ_m (L[i][m] + N[i][m]) · E[m+1][j]

最后一行是整道题的灵魂，值得单独说：

E 只能停在"不带星尾的块"上。 (X)S 后面必须再接一块，否则这个串以 * 结尾，不是合法串。

我第一版写成了 E[i][j] = (L+N)[i][j] + Σ …，于是 (*)**、(**)*** 这种以星结尾的串全被算进去了。

按长度从小到大递推即可。L 的两个求和是 O(n)，N、E 各一个求和 O(n)，总复杂度 O(n³)。

第一档：15 分（测试点 1~3，n ≤ 15）

n 只有 15，直接枚举每个 ? 填什么，然后用题面的三条规则本身判断合法性（递归 + 记忆化）。

为了不在 3n 里淹死，加四条必要剪枝（都是合法串必须满足的）：

  1. 任意前缀中 ( 的个数不少于 ) 的个数；
  2. 连续 * 不超过 k 个；
  3. 第一个字符不能是 )，最后一个字符必须是 )；
  4. 剩下的位置不够把未匹配的 ( 全部闭合时直接回溯。

实测 n = 15 全 ?、k = 3 时只要 0.12s，k = 15 时 0.14s。这份代码和后面的正解、以及最朴素的暴力三方对拍过 500+ 组。

C++
// CSP-S 2021 T2 括号序列 —— 15 分档（测试点 1~3，n <= 15）
// 做法：直接枚举每个 '?' 填 '('/')'/'*'，每得到完整串用"题面 3 条规则"的递归判定检查。
// 为了不真的跑到 3^n，加了 4 条必要剪枝（都是合法串必须满足的）：
//   1. 任意前缀里 '(' 的个数不少于 ')' 的个数；
//   2. 连续的 '*' 不超过 k 个；
//   3. 第一个字符不能是 ')'，最后一个字符必须是 ')'；
//   4. 剩下的位置不够把未匹配的 '(' 全部闭合时直接回溯。
// 判定函数 V(l,r) 直接照抄题面规则（带记忆化），与后面的 DP 完全独立。
#include <bits/stdc++.h>
using namespace std;

int n, K;
char s[64], t[64];
long long ans = 0;
int vis[64][64];
bool val[64][64];

inline bool isStar(int l, int r) {
    if (l > r || r - l + 1 > K) return false;
    for (int i = l; i <= r; i++) if (t[i] != '*') return false;
    return true;
}

// [l,r] 是否是"符合规范的超级括号序列"
bool V(int l, int r) {
    if (l > r) return false;
    if (vis[l][r]) return val[l][r];
    vis[l][r] = 1;
    bool res = false;
    if (r == l + 1 && t[l] == '(' && t[r] == ')') res = true;                  // ()
    if (!res && t[l] == '(' && t[r] == ')' && isStar(l + 1, r - 1)) res = true; // (S)
    if (!res)                                                                  // AB
        for (int m = l; m < r && !res; m++) if (V(l, m) && V(m + 1, r)) res = true;
    if (!res)                                                                  // ASB
        for (int m1 = l; m1 < r && !res; m1++) {
            if (!V(l, m1)) continue;
            for (int m2 = m1 + 1; m2 < r; m2++)
                if (isStar(m1 + 1, m2 - 1) && V(m2, r)) { res = true; break; }
        }
    if (!res && t[l] == '(' && t[r] == ')' && V(l + 1, r - 1)) res = true;      // (A)
    if (!res && t[l] == '(' && t[r] == ')')                                    // (SA)
        for (int m = l + 1; m < r && !res; m++) if (isStar(l + 1, m) && V(m + 1, r - 1)) res = true;
    if (!res && t[l] == '(' && t[r] == ')')                                    // (AS)
        for (int m = l + 1; m < r && !res; m++) if (V(l + 1, m) && isStar(m + 1, r - 1)) res = true;
    val[l][r] = res;
    return res;
}

void dfs(int p, int bal, int star) {   // bal: 未匹配的 '(' 数, star: 当前连续 '*' 数
    if (bal > n - p) return;                         // 剩下的位置不够闭合
    if (p == n) {
        if (bal != 0 || t[n - 1] != ')') return;     // 合法串必须以 ')' 结尾
        memset(vis, 0, sizeof vis);
        if (V(0, n - 1)) ans++;
        return;
    }
    char lo = '(', hi = ')';
    char cand[3] = { '(', ')', '*' };
    for (int id = 0; id < 3; id++) {
        char c = cand[id];
        if (s[p] != '?' && s[p] != c) continue;
        if (p == 0 && c == ')') continue;            // 不能以 ')' 开头
        if (c == ')' && bal == 0) continue;          // 无括号可匹配
        if (c == '*' && star + 1 > K) continue;      // 星串超长
        t[p] = c;
        dfs(p + 1, c == '(' ? bal + 1 : (c == ')' ? bal - 1 : bal), c == '*' ? star + 1 : 0);
    }
}

int main() {
    if (scanf("%d %d %s", &n, &K, s) != 3) return 0;
    dfs(0, 0, 0);
    printf("%lld\n", ans % 1000000007LL);
    return 0;
}

第二档：65 分（测试点 1~13，n ≤ 100）

状态定义和正解完全一样，只是把 E 的转移写成"第一块 [i,a] + 中间星串 [a+1,b] + 剩余 [b+1,j]"的三层枚举：

文本
for a in [i, j-1]:
  for b in [a, j-1]:
     E[i][j] += L[i][a] · (b==a ? 1 : Sg[a+1][b]) · E[b+1][j]

复杂度 O(n⁴)。n ≤ 100 时约 4 × 106，实测 0.01s；n = 500 时理论 2.6 × 109，实测 1.73s —— 超过 1s 时限，所以这份代码只能拿 65 分。

（顺带说：为什么 14~15 这两个"全是 ?"的点也过不了？因为它们的 n 也是 500，和最后 5 个点一样被卡在复杂度上。）

C++
// CSP-S 2021 T2 括号序列 —— 65 分档（测试点 1~13，n <= 100）
// 状态定义与正解完全相同，只是 E 的转移写成"第一块 [i,a] + 中间星串 [a+1,b] + 剩余 [b+1,j]"的三层枚举，
// 复杂度 O(n^4)。n <= 100 时约 4e6 次，n = 500 时约 2.6e9 次，会超时。
#include <bits/stdc++.h>
using namespace std;

const int MOD = 1000000007;
int n, K;
char s[505];
long long E[505][505], L[505][505];
bool Sg[505][505];

inline bool cOpen(char c) { return c == '(' || c == '?'; }
inline bool cClose(char c) { return c == ')' || c == '?'; }
inline bool cStar(char c) { return c == '*' || c == '?'; }

int main() {
    if (scanf("%d %d %s", &n, &K, s + 1) != 3) return 0;

    for (int i = 1; i <= n; i++) {
        bool all = true;
        for (int j = i; j <= n; j++) {
            if (!cStar(s[j])) all = false;
            Sg[i][j] = all && (j - i + 1 <= K);
        }
    }

    for (int len = 1; len <= n; len++) {
        for (int i = 1; i + len - 1 <= n; i++) {
            int j = i + len - 1;

            // ---- L[i][j]：一对括号恰好包住 [i,j] ----
            long long l = 0;
            if (len >= 2 && cOpen(s[i]) && cClose(s[j])) {
                int a = i + 1, b = j - 1;
                if (len == 2) l = 1;
                else {
                    l += Sg[a][b];
                    l += E[a][b];
                    for (int m = a; m < b; m++) {
                        l += Sg[a][m] * E[m + 1][b] % MOD;
                        l += E[a][m] * Sg[m + 1][b] % MOD;
                    }
                }
                l %= MOD;
            }
            L[i][j] = l;

            // ---- E[i][j]：三元枚举（块 / 星串 / 剩余）----
            long long e = L[i][j];
            for (int a = i; a < j; a++) {
                if (!L[i][a]) continue;
                for (int b = a; b < j; b++) {
                    long long mid;
                    if (b == a) mid = 1;
                    else mid = Sg[a + 1][b] ? 1 : 0;
                    if (!mid) continue;
                    if (!E[b + 1][j]) continue;
                    e += L[i][a] * mid % MOD * E[b + 1][j] % MOD;
                }
            }
            E[i][j] = e % MOD;
        }
    }
    printf("%lld\n", E[1][n] % MOD);
    return 0;
}

第三档：100 分（n ≤ 500）

把 E 的转移从 O(n) 降回 O(n)：

文本
E[i][j] = L[i][j] + Σ_m (L+N)[i][m] · E[m+1][j]

总复杂度 O(n³) ≈ 500³/6 × 4 ≈ 8 × 107，实测 n = 500 全 ? 只要 0.05s。

C++
// CSP-S 2021 T2 括号序列 —— 100 分（n <= 500）
// 关键观察：任何"符合规范的超级括号序列"都以 '(' 开头、以 ')' 结尾。
//   所以整串可以唯一地切成若干"块"，每块形如  ( X )  或  ( X ) S
//   其中 X ∈ { 空, S, A, SA, AS }（A 为合法串，S 为非空且长度 <= k 的星串）。
// 记：
//   Sg[i][j] : [i,j] 本身是一个合法星串（非空、长度 <= k、全是 '*' 或 '?'）
//   L[i][j]  : [i,j] 恰好是"一块括号" ( X )
//   N[i][j]  : [i,j] 是"一块括号 + 尾部星串" ( X ) S
//   P[i][j]  : L + N（"一个块"，可能带星尾）
//   E[i][j]  : [i,j] 是合法串（若干块拼起来）
// 转移见下方注释。复杂度 O(n^3)。
#include <bits/stdc++.h>
using namespace std;

const int MOD = 1000000007;
int n, K;
char s[505];
long long E[505][505], L[505][505], Nn[505][505], P[505][505];
bool Sg[505][505];

inline bool cOpen(char c) { return c == '(' || c == '?'; }
inline bool cClose(char c) { return c == ')' || c == '?'; }
inline bool cStar(char c) { return c == '*' || c == '?'; }

int main() {
    if (scanf("%d %d %s", &n, &K, s + 1) != 3) return 0;

    // 预处理星串
    for (int i = 1; i <= n; i++) {
        bool all = true;
        for (int j = i; j <= n; j++) {
            if (!cStar(s[j])) all = false;
            Sg[i][j] = all && (j - i + 1 <= K);
        }
    }

    for (int len = 1; len <= n; len++) {
        for (int i = 1; i + len - 1 <= n; i++) {
            int j = i + len - 1;
            long long l = 0, nn = 0, e = 0;

            // ---- L[i][j]：最外层一对括号恰好包住 [i,j] ----
            if (len >= 2 && cOpen(s[i]) && cClose(s[j])) {
                int a = i + 1, b = j - 1;
                if (len == 2) l = 1;                       // ()
                else {
                    l += Sg[a][b];                         // (S)
                    l += E[a][b];                          // (A)
                    for (int m = a; m < b; m++) {
                        l += Sg[a][m] * E[m + 1][b] % MOD;  // (SA)：前面星串，后面合法串
                        l += E[a][m] * Sg[m + 1][b] % MOD;  // (AS)：前面合法串，后面星串
                    }
                }
                l %= MOD;
            }
            L[i][j] = l % MOD;

            // ---- N[i][j]：(X) 后接一段非空星串 ----
            for (int m = i + 1; m < j; m++)
                if (L[i][m] && Sg[m + 1][j])
                    nn += L[i][m] * Sg[m + 1][j] % MOD;
            Nn[i][j] = nn % MOD;

            P[i][j] = (L[i][j] + Nn[i][j]) % MOD;

            // ---- E[i][j]：合法串必须以 ')' 结尾，所以不能停在"带星尾的块"上 ----
            // 只有单个括号块 L 可以直接结束；(X)S 后面必须再接块
            e = L[i][j];
            for (int m = i; m < j; m++)
                if (P[i][m] && E[m + 1][j])
                    e += P[i][m] * E[m + 1][j] % MOD;
            E[i][j] = e % MOD;
        }
    }
    printf("%lld\n", E[1][n] % MOD);
    return 0;
}

易错点

  1. E 不能以带星尾的块结尾（前面踩过）。
  2. () 的长度是 2，L 里要特判 len == 2；L 对 len == 1 恒为 0。
  3. (SA) 和 (AS) 的拆分点唯一（S 必须吃掉全部连续的 *，因为 A 必须以 ( 开头 / 以 ) 结尾），所以不会重复计数；但如果你把条件写松（比如 S 允许空），就会重复。
  4. Sg 的"非空"和"长度 ≤ k"两个条件都要卡，(S) 里的 S 是非空的。
  5. 全部乘法都要 % MOD，long long 存中间值。
  6. 样例 2（???(*??(?, k=2）答案是 19，样例 1 是 5。这两个样例能过，基本说明状态没漏；但强烈建议再写个暴力对拍，因为状态漏一个只会在特定数据上暴露。

T3 · 回文

题意

给长度 2n 的序列，1..n 每个数恰好出现两次。每次可以从当前序列的开头或末尾取一个数，依次放进 b。问能否让 b 成为回文；若能，输出长度为 2n 的 L/R 操作串中字典序最小的一个。

数据范围与子任务

测试点 分值 累计 T ≤ n ≤ Σn ≤ 特殊性质
1 ~ 7 28 28 10 10 50 无
8 ~ 10 12 40 100 20 1000 无
11 ~ 12 8 48 100 100 1000 无
13 ~ 15 12 60 100 1000 25000 无
16 ~ 17 8 68 1 5×105 5×105 无
18 ~ 20 12 80 100 5×105 5×105 有
21 ~ 25 20 100 100 5×105 5×105 无

"特殊性质"：每次删除两个相邻且相等的数，存在一种方式把序列删空（例如 a = [1,2,2,1]）。

三个观察

观察 1（起点）：第一次操作只能取 a[1] 或 a[2n]。两种都试一遍，取字典序小的可行解。

观察 2（终点）：设第一次取到的数是 v，它在序列里的另一个位置是 pos。因为 b 回文，b[2n] = b[1] = v，而 v 只剩 pos 这一处，所以第 2n 次操作必然取 pos。

于是序列被 v 的两次出现切成两段：

文本
L 段 = [1, pos-1]     R 段 = [pos+1, 2n]      （下标按第一步取 a[1] 说明）

观察 3（同时定两端）：第 i 次操作与第 2n+1−i 次操作取的数必须相等（回文的定义）。

  • 第 i 次（i ≤ n，属于前半段）只能在 L 的左端 或 R 的右端取 —— 因为 L 段只能从左往右剥，R 段只能从右往左剥；
  • 第 2n+1−i 次（属于后半段）只能在 L 的右端 或 R 的左端取。

两者相等，于是每轮有四种组合，按字典序（先比第 i 位，再比第 2n+1−i 位；L < R）依次尝试：

优先级 第 i 次 第 2n+1−i 次 输出两位 指针变化
1 L 左端 L 右端 L L l1++, r1--
2 L 左端 R 左端 L R l1++, l2++
3 R 右端 L 右端 R L r2--, r1--
4 R 右端 R 左端 R R r2--, l2++

关键一行：输出字符由"这个数属于哪一段"决定，不是由"从左还是从右"决定。

  • 取 L 段的任何元素 → L（L 段元素只能从左边剥到）；
  • 取 R 段的任何元素 → R（R 段元素只能从右边剥到）。

这一点我第一次写错了：把"取 R 段左端"记成了 L，于是样例输出 LRRLLLLLLL（正确答案 LRRLLRRRRL）。原因是 R 段的左端虽然是"位置上靠左的元素"，但 pos 挡在它左边，它只能从右端被剥到。

第 2n 次（最后取 pos）时只剩一个元素，记 L（字典序更小）。

为什么按字典序贪心是对的

  • 每轮的四种组合里，能成立的至多一种"必要"的：因为每段两端各只有一个数，一旦要求"第 i 次 = 第 2n+1−i 次"，能配上的数对是确定的。
  • 若真的有两种组合同时成立（例如 L 左端 = L 右端 = R 左端），选哪个？题解的说法是：因为每个数恰好出现两次，两个操作互不影响 —— 一对数一旦被配对消耗掉，就不会再参与后面的匹配，所以先选字典序小的那个不会"抢走"后面要用的资源。
  • 两个起点（第一步 L / 第一步 R）都试一遍，取较小的那个。

第一档：28 分（测试点 1~7，n ≤ 10）

直接枚举 22n 个操作串，逐个模拟得到 b，检查回文，取字典序最小的。

n = 10 时 220 ≈ 106，每个 O(n) 检查，T = 10 时约 2 × 107，实测 0.5s。

C++
// CSP-S 2021 T3 回文 —— 28 分档（测试点 1~7，n <= 10）
// 直接枚举全部 2^(2n-1) 种操作串，模拟出 b 并检查是否回文，取字典序最小者。
// n <= 10 时 2^19 = 5e5，配 T <= 10 也能跑。
#include <bits/stdc++.h>
using namespace std;

int n, a[64];
string best;
bool has;

void check(const string &ops) {
    int l = 0, r = 2 * n - 1;
    static int b[64];
    for (int i = 0; i < 2 * n; i++) b[i] = (ops[i] == 'L') ? a[l++] : a[r--];
    for (int i = 0; i < n; i++) if (b[i] != b[2 * n - 1 - i]) return;
    if (!has || ops < best) { best = ops; has = true; }
}

int main() {
    int T;
    if (scanf("%d", &T) != 1) return 0;
    while (T--) {
        scanf("%d", &n);
        for (int i = 0; i < 2 * n; i++) scanf("%d", &a[i]);
        has = false; best.clear();
        string ops(2 * n, 'L');
        for (long long m = 0; m < (1LL << (2 * n)); m++) {
            for (int i = 0; i < 2 * n; i++) ops[i] = (m >> i & 1) ? 'R' : 'L';
            check(ops);
        }
        if (!has) puts("-1");
        else puts(best.c_str());
    }
    return 0;
}

第二档：60 分（测试点 8~15，n ≤ 1000）

思路已经是正解（四指针同时定前后两次操作），但数据结构选错了：用 vector 存两段，每次取数 erase(begin()) 是 O(n) 的，总复杂度退化成 O(n²)。

n ≤ 1000 时约 106 次搬移，秒过；n = 5 × 105 时约 2.5 × 1011，实测 14.65s，必然超时。

这是考场上非常典型的失分原因：算法对了，容器选错。（改成 deque，或者干脆用下标就是 O(n)，见第三档。）

C++
// CSP-S 2021 T3 回文 —— 60 分档（测试点 1~15，n <= 1000）
// 思路与正解完全一致（四指针同时定第 i / 第 2n+1-i 次操作），
// 但用 vector 保存两段并在取数时 erase(begin())，单次 O(n) -> 总 O(n^2)。
// n <= 1000 时约 1e6 次搬移，秒过；n = 5e5 时约 2.5e11，必然超时。
#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> A;
vector<int> other_pos;

bool build(vector<int> L, vector<int> R, char first, string &ans) {
    ans.assign(2 * n, '?');
    ans[0] = first;
    ans[2 * n - 1] = 'L';
    for (int i = 1; i < n; i++) {
        int j = 2 * n - 1 - i;
        bool okL = !L.empty(), okR = !R.empty();
        if (L.size() >= 2 && L.front() == L.back()) {                    // L 左端 + L 右端
            ans[i] = 'L'; ans[j] = 'L';
            L.erase(L.begin()); L.pop_back();
        } else if (okL && okR && L.front() == R.front()) {               // L 左端 + R 左端
            ans[i] = 'L'; ans[j] = 'R';
            L.erase(L.begin()); R.erase(R.begin());
        } else if (okL && okR && R.back() == L.back()) {                 // R 右端 + L 右端
            ans[i] = 'R'; ans[j] = 'L';
            R.pop_back(); L.pop_back();
        } else if (R.size() >= 2 && R.back() == R.front()) {             // R 右端 + R 左端
            ans[i] = 'R'; ans[j] = 'R';
            R.pop_back(); R.erase(R.begin());
        } else return false;
    }
    return true;
}

void solve() {
    scanf("%d", &n);
    A.assign(2 * n, 0);
    for (int i = 0; i < 2 * n; i++) scanf("%d", &A[i]);

    vector<int> firstpos(n + 1, -1);
    other_pos.assign(2 * n, -1);
    for (int i = 0; i < 2 * n; i++) {
        if (firstpos[A[i]] == -1) firstpos[A[i]] = i;
        else { other_pos[firstpos[A[i]]] = i; other_pos[i] = firstpos[A[i]]; }
    }

    string ans1, ans2;
    bool ok1 = false, ok2 = false;
    {
        int pos = other_pos[0];
        if (pos > 0) {
            vector<int> L(A.begin() + 1, A.begin() + pos);
            vector<int> R(A.begin() + pos + 1, A.end());
            ok1 = build(L, R, 'L', ans1);
        }
    }
    {
        int pos = other_pos[2 * n - 1];
        if (pos < 2 * n - 1) {
            vector<int> L(A.begin(), A.begin() + pos);
            vector<int> R(A.begin() + pos + 1, A.end() - 1);
            ok2 = build(L, R, 'R', ans2);
        }
    }
    if (!ok1 && !ok2) puts("-1");
    else if (ok1 && ok2) puts((ans1 < ans2 ? ans1 : ans2).c_str());
    else puts((ok1 ? ans1 : ans2).c_str());
}

int main() {
    int T; scanf("%d", &T);
    while (T--) solve();
    return 0;
}

第三档：100 分（n ≤ 5×105）

把两段换成四个下标 [l1,r1]、[l2,r2]，全程只做 l1++/r1--/l2++/r2--，O(1) 一次。

两个起点各跑一遍，用 strcmp 比较取较小的（不要用 std::string 拼接，5 × 105 长度下会慢）。

复杂度 O(n)，实测 n = 5 × 105 时 0.04s。

C++
// CSP-S 2021 T3 回文 —— 100 分（n <= 5e5, sum n <= 5e5）
// 观察：第一个取的数 v 一定是 a[1] 或 a[2n]；它的另一次出现位置 pos 必然是最后一个取的数。
// 于是 v 的两次出现把整个序列切成左右两段 L = [l1,r1]、R = [l2,r2]。
// 之后每一步同时确定"第 i 次"和"第 2n+1-i 次"两个操作：
//   第 i 次只能在 L 的左端或 R 的右端取；第 2n+1-i 次只能在 L 的右端或 R 的左端取；
//   两者数值必须相同。四种组合按字典序（先比第 i 位，再比第 2n+1-i 位）依次尝试：
//     (L左, L右) -> "LL"    (L左, R左) -> "LR"    (R右, L右) -> "RL"    (R右, R左) -> "RR"
// 取 L 段任何元素都是"从左边剥"记 'L'，取 R 段任何元素都是"从右边剥"记 'R'。
// 两个起点各跑一遍，取字典序小的答案。复杂度 O(n)。
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 500005;
int n, a[2 * MAXN];
int other_pos[2 * MAXN];
char ans[2 * MAXN + 1];

bool build(int l1, int r1, int l2, int r2, char first) {
    ans[0] = first;
    ans[2 * n - 1] = 'L';
    for (int i = 1; i < n; i++) {
        int j = 2 * n - 1 - i;
        if (l1 < r1 && a[l1] == a[r1]) {                      // (L左端, L右端) -> L L
            ans[i] = 'L'; ans[j] = 'L'; l1++; r1--;
        } else if (l1 <= r1 && l2 <= r2 && a[l1] == a[l2]) {   // (L左端, R左端) -> L R
            ans[i] = 'L'; ans[j] = 'R'; l1++; l2++;
        } else if (l1 <= r1 && l2 <= r2 && a[r2] == a[r1]) {   // (R右端, L右端) -> R L
            ans[i] = 'R'; ans[j] = 'L'; r2--; r1--;
        } else if (l2 < r2 && a[r2] == a[l2]) {                // (R右端, R左端) -> R R
            ans[i] = 'R'; ans[j] = 'R'; r2--; l2++;
        } else return false;
    }
    ans[2 * n] = 0;
    return true;
}

int main() {
    int T;
    if (scanf("%d", &T) != 1) return 0;
    while (T--) {
        scanf("%d", &n);
        for (int i = 0; i < 2 * n; i++) scanf("%d", &a[i]);
        for (int i = 0; i < 2 * n; i++) other_pos[i] = -1;
        static int firstpos[MAXN];
        for (int v = 1; v <= n; v++) firstpos[v] = -1;
        for (int i = 0; i < 2 * n; i++) {
            if (firstpos[a[i]] == -1) firstpos[a[i]] = i;
            else { other_pos[firstpos[a[i]]] = i; other_pos[i] = firstpos[a[i]]; }
        }
        static char buf1[2 * MAXN + 1], buf2[2 * MAXN + 1];
        bool ok1 = false, ok2 = false;
        {
            int pos = other_pos[0];
            if (pos > 0 && build(1, pos - 1, pos + 1, 2 * n - 1, 'L')) { ok1 = true; memcpy(buf1, ans, 2 * n + 1); }
        }
        {
            int pos = other_pos[2 * n - 1];
            if (pos < 2 * n - 1 && build(0, pos - 1, pos + 1, 2 * n - 2, 'R')) { ok2 = true; memcpy(buf2, ans, 2 * n + 1); }
        }
        if (!ok1 && !ok2) puts("-1");
        else if (ok1 && ok2) puts(strcmp(buf1, buf2) <= 0 ? buf1 : buf2);
        else puts(ok1 ? buf1 : buf2);
    }
    return 0;
}

易错点

  1. 最后一位固定是 L：剩单个元素时，L 和 R 指同一个位置，取字典序更小的 L。
  2. 输出字符按"段"决定，不按"方向"决定（前面踩过，这是本题最常见的 WA）。
  3. L.size() >= 2 才能走"L 左端 + L 右端"（同一个元素不能和自己配对）；R 同理。
  4. 两段都空时循环必须已经结束：循环跑 n-1 轮，正好消耗 2(n-1) 个数。
  5. 两个起点都可行时要比字典序；只有一个可行就直接输出；都不可行输出 -1。
  6. 多组数据，每组的 firstpos 数组要重新清空（T ≤ 100、值域是 1..n，只清 1..n 即可）。

T4 · 交通规划

题意

n 条横线、m 条竖线交出 n × m 个格点，相邻格点之间的边有非负权。网格外有 2n+2m 条射线（按"左上→右上→右下→左下"编号），每次询问在其中 k 条射线上各放一个附加点，给出它到最近格点的边权 x3 和颜色（黑/白）。

给所有格点黑白染色，代价 = 所有"两端异色"的边的边权和（包括附加点那条边）。求最小代价。

第一步：先把它看成一个最小割

给每个格点一个 0/1 变量。每条网格边 (u,v,w) 贡献 w · [xu ≠ xv]；每个附加点（颜色 c、边权 w）贡献 w · [xu ≠ c]。

这是标准的二值最小割：

  • 源 S 表示"黑"，汇 T 表示"白"；
  • 每条网格边加两条有向边 u→v、v→u，容量都是 w；
  • 黑色附加点：S → u 容量 w（u 若被染白，这条边就被割）；
  • 白色附加点：u → T 容量 w（u 若被染黑，这条边就被割）。

答案 = 最小割 = 最大流。

第一档：40 分（测试点 1~8，n,m ≤ 18）

直接跑 Dinic。n,m ≤ 18 时结点数 ≤ 324，单次最大流非常快，实测 k = 50 交替色只要 0.01s。

这份代码在 n,m ≤ 18 的 8 个点（10 + 15 + 15 = 40 分）上都能过；n,m = 100 时结点 104、边 4 × 104，Dinic 在网格图上会退化，跑不动。

C++
// CSP-S 2021 T4 交通规划 —— 40 分档（测试点 1~8，n,m <= 18）
// 直接建最小割模型：
//   把每个格点当结点，源 S 代表"黑"，汇 T 代表"白"。
//   对每条网格边 (u,v,w) 加两条有向边 u->v、v->u 容量 w；
//   黑附加点：S -> u 容量 w（u 若被染白，这条边就被割）
//   白附加点：u -> T 容量 w（u 若被染黑，这条边就被割）
//   答案就是 S-T 最小割 = 最大流。
// n,m <= 18 时结点数 <= 324，单次 Dinic 足够快。
#include <bits/stdc++.h>
using namespace std;

const int MAXV = 400;
const int MAXE = 8000;
int n, m, T;
int X1[32][32], X2[32][32];
int rayNode(int p) {                       // 射线 -> 格点（0-indexed）
    if (p <= m) return (p - 1);
    if (p <= m + n) return (p - m - 1) * m + (m - 1);
    if (p <= 2 * m + n) { int i = p - (m + n); return (n - 1) * m + (m - i); }
    int i = p - (2 * m + n); return (n - i) * m;
}

int head[MAXV], nxt[MAXE], to[MAXE], cap[MAXE], ecnt;
inline void addEdge(int u, int v, int c) {
    to[ecnt] = v; cap[ecnt] = c; nxt[ecnt] = head[u]; head[u] = ecnt++;
    to[ecnt] = u; cap[ecnt] = 0; nxt[ecnt] = head[v]; head[v] = ecnt++;
}
inline void addUndir(int u, int v, int c) {   // 双向各 c
    to[ecnt] = v; cap[ecnt] = c; nxt[ecnt] = head[u]; head[u] = ecnt++;
    to[ecnt] = u; cap[ecnt] = c; nxt[ecnt] = head[v]; head[v] = ecnt++;
}

int level[MAXV], iter[MAXV];
bool bfs(int s, int t, int V) {
    for (int i = 0; i < V; i++) level[i] = -1;
    queue<int> q; q.push(s); level[s] = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int e = head[u]; e != -1; e = nxt[e])
            if (cap[e] > 0 && level[to[e]] < 0) { level[to[e]] = level[u] + 1; q.push(to[e]); }
    }
    return level[t] >= 0;
}
int dfs(int u, int t, int f) {
    if (u == t) return f;
    for (int &e = iter[u]; e != -1; e = nxt[e]) {
        int v = to[e];
        if (cap[e] > 0 && level[v] == level[u] + 1) {
            int d = dfs(v, t, min(f, cap[e]));
            if (d > 0) { cap[e] -= d; cap[e ^ 1] += d; return d; }
        }
    }
    return 0;
}
long long dinic(int s, int t, int V) {
    long long flow = 0;
    while (bfs(s, t, V)) {
        for (int i = 0; i < V; i++) iter[i] = head[i];
        int f;
        while ((f = dfs(s, t, INT_MAX)) > 0) flow += f;
    }
    return flow;
}

int main() {
    if (scanf("%d %d %d", &n, &m, &T) != 3) return 0;
    for (int i = 1; i <= n - 1; i++) for (int j = 1; j <= m; j++) scanf("%d", &X1[i][j]);
    for (int i = 1; i <= n; i++) for (int j = 1; j <= m - 1; j++) scanf("%d", &X2[i][j]);

    while (T--) {
        int k; scanf("%d", &k);
        int V = n * m + 2, S = n * m, TT = n * m + 1;
        ecnt = 0;
        for (int i = 0; i < V; i++) head[i] = -1;
        for (int i = 1; i <= n - 1; i++) for (int j = 1; j <= m; j++)
            addUndir((i - 1) * m + (j - 1), i * m + (j - 1), X1[i][j]);
        for (int i = 1; i <= n; i++) for (int j = 1; j <= m - 1; j++)
            addUndir((i - 1) * m + (j - 1), (i - 1) * m + j, X2[i][j]);
        for (int i = 0; i < k; i++) {
            int x3, p, c; scanf("%d %d %d", &x3, &p, &c);
            int u = rayNode(p);
            if (c == 1) addEdge(S, u, x3);     // 黑：源侧
            else addEdge(u, TT, x3);           // 白：汇侧
        }
        printf("%lld\n", dinic(S, TT, V));
    }
    return 0;
}

第二步：平面图最小割 = 对偶图最短路

网格图是平面图，最小割 = 对偶图上的一条路径。对偶图的建法：

  1. 每个小方格是一个"内部面"，共 (n−1)(m−1) 个结点；
  2. 原图的每条边对应一条对偶边，权不变：上下相邻的面之间用"横向边"的权，左右相邻的面之间用"纵向边"的权；
  3. 外部面（网格外面那一圈）被 k 个附加点所在的射线切成 k 个扇区：
    • 把附加点按射线编号排序，扇区 s 从射线 P[s] 顺时针走到射线 P[s+1]；
    • 这段路径上经过的每条边界边，都在对偶图里连一条边到"边界内侧的那个内部面"，权 = 边界边权；
    • 射线本身不是原图的边，所以扇区与扇区之间没有边，除了下面这一条：
  4. 相邻扇区之间隔着一条射线，该射线上有附加点，就加一条边，权 = 该附加点的 x3。它的含义是"直接把附加点那条边剪断"。

k = 2 时：答案就是扇区 S0 到扇区 S1 的最短路。三种可能都包含在里面了：

  • 两个附加点同色 → 全染同色，答案是 0（代码里特判）；
  • 异色 → 最短路要么穿过网格内部，要么直接走"剪断某个附加点"的那条边（对应"全染成另一种颜色"）。

位置映射（本题最容易写错的地方）

射线是沿边界顺时针编号的，四个角上的格点会被算两次（比如右上角既是"上边的最后一个"，又是"右边的第一个"）。所以 2n+2m 条射线之间只有 2n+2m 个"位置"，其中 4 个位置是角点、没有边界边，剩下 2n+2m−4 个位置各对应一条边界边。

设位置 p 表示"射线 p 和射线 p+1 之间"的那段边界，用 0-indexed：

位置 p 范围 对应边界边 内侧的面 权
0 ~ m−2 (1, p+1) — (1, p+2) (1, p+1) X2[1][p+1]
m−1 角点 (1,m)，无边 — —
m ~ m+n−2 (i, m) — (i+1, m)，i = p−m+1 (i, m−1) X1[i][m]
m+n−1 角点 (n,m) — —
m+n ~ 2m+n−2 (n, c−1) — (n, c)，c = 2m+n−p (n−1, c−1) X2[n][c−1]
2m+n−1 角点 (n,1) — —
2m+n ~ 2m+2n−2 (r−1, 1) — (r, 1)，r = 2m+2n−p (r−1, 1) X1[r−1][1]
2m+2n−1 角点 (1,1) — —

（X1[i][j] 是 (i,j)—(i+1,j) 的竖直边，X2[i][j] 是 (i,j)—(i,j+1) 的水平边。）

第二档：70 分（k ≤ 2 的 7 个测试点 + 40 分的暴力部分）

只处理 k ≤ 2：建对偶图，两个扇区各连上自己覆盖的边界边，再加上两条"剪断附加点"的边，跑一次 Dijkstra。

k = 1 或两点同色时答案是 0。

C++
// CSP-S 2021 T4 交通规划 —— 70 分档（k <= 2 的 7 个测试点 + n,m <= 18 的暴力分）
// 思路：k = 2 时若两点同色，全染同色，答案 0；若异色，问题就是"把两个附加点分开的最小割"。
// 平面图最小割 = 对偶图最短路：
//   内部面 (n-1)(m-1) 个结点，对偶边权 = 原边权；
//   外部面被两条射线切成两个扇区 S0、S1，扇区覆盖的每条边界边都连一条边到边界内侧的面；
//   S0、S1 之间再连一条边（权 = 射线上附加点的 x3），表示"直接剪断附加点那条边"。
//   答案 = S0 到 S1 的最短路。复杂度 O(nm log(nm))。
// 注意：这份代码只处理 k <= 2（那 7 个测试点），k > 2 的情况请见 100 分档。
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 505;
const int MAXV = MAXN * MAXN + 64;
const long long INF = (long long)1e18;

int n, m, T;
int X1[MAXN][MAXN], X2[MAXN][MAXN];
int V0;
struct E { int to, w; };
vector<E> g[MAXV];
inline void addE(int u, int v, int w) { g[u].push_back({v, w}); g[v].push_back({u, w}); }
inline int fid(int i, int j) { return (i - 1) * (m - 1) + j; }

int posF[MAXN * 4], posW[MAXN * 4];
long long dist[MAXV];

void dijkstra(int s, int V) {
    for (int i = 1; i <= V; i++) dist[i] = INF;
    priority_queue<pair<long long, int>, vector<pair<long long, int> >, greater<pair<long long, int> > > pq;
    dist[s] = 0; pq.push({0, s});
    while (!pq.empty()) {
        pair<long long, int> t = pq.top(); pq.pop();
        int u = t.second;
        if (dist[u] < t.first) continue;
        for (size_t i = 0; i < g[u].size(); i++) {
            int v = g[u][i].to, w = g[u][i].w;
            if (dist[u] + w < dist[v]) { dist[v] = dist[u] + w; pq.push({dist[v], v}); }
        }
    }
}

int main() {
    if (scanf("%d %d %d", &n, &m, &T) != 3) return 0;
    for (int i = 1; i <= n - 1; i++) for (int j = 1; j <= m; j++) scanf("%d", &X1[i][j]);
    for (int i = 1; i <= n; i++) for (int j = 1; j <= m - 1; j++) scanf("%d", &X2[i][j]);

    V0 = (n - 1) * (m - 1);
    for (int i = 1; i <= n - 1; i++)
        for (int j = 1; j <= m - 1; j++) {
            if (i + 1 <= n - 1) addE(fid(i, j), fid(i + 1, j), X2[i + 1][j]);
            if (j + 1 <= m - 1) addE(fid(i, j), fid(i, j + 1), X1[i][j + 1]);
        }

    int N = 2 * m + 2 * n;
    for (int p = 0; p < N; p++) { posF[p] = 0; posW[p] = 0; }
    for (int p = 0; p <= m - 2; p++) { posF[p] = fid(1, p + 1); posW[p] = X2[1][p + 1]; }
    for (int p = m; p <= m + n - 2; p++) { int i = p - m + 1; posF[p] = fid(i, m - 1); posW[p] = X1[i][m]; }
    for (int p = m + n; p <= 2 * m + n - 2; p++) { int c = 2 * m + n - p; posF[p] = fid(n - 1, c - 1); posW[p] = X2[n][c - 1]; }
    for (int p = 2 * m + n; p <= 2 * m + 2 * n - 2; p++) { int r = 2 * m + 2 * n - p; posF[p] = fid(r - 1, 1); posW[p] = X1[r - 1][1]; }

    vector<int> touched;
    while (T--) {
        // 清掉上一轮挂在内部面上的动态边（扇区结点自己的出边也一并清掉）
        for (size_t i = 0; i < touched.size(); i++) {
            int f = touched[i];
            size_t w = 0;
            for (size_t j = 0; j < g[f].size(); j++)
                if (g[f][j].to <= V0) g[f][w++] = g[f][j];
            g[f].resize(w);
        }
        touched.clear();
        g[V0 + 1].clear(); g[V0 + 2].clear();

        int k; scanf("%d", &k);
        vector<int> P(k), C(k), W(k);
        for (int i = 0; i < k; i++) { scanf("%d %d %d", &W[i], &P[i], &C[i]); P[i]--; }
        if (k == 1) { puts("0"); continue; }
        if (k == 2) {
            // 按射线编号排好序，保证环序
            if (P[0] > P[1]) { swap(P[0], P[1]); swap(C[0], C[1]); swap(W[0], W[1]); }
            if (C[0] == C[1]) { puts("0"); continue; }
            int V = V0 + 2;
            int a = P[0], b = P[1];
            int t = a;
            while (t != b) { if (posF[t]) { addE(V0 + 1, posF[t], posW[t]); touched.push_back(posF[t]); } t = (t + 1) % N; }
            t = b;
            while (t != a) { if (posF[t]) { addE(V0 + 2, posF[t], posW[t]); touched.push_back(posF[t]); } t = (t + 1) % N; }
            addE(V0 + 1, V0 + 2, W[1]);   // 经过射线 P[1] 上的附加点
            addE(V0 + 1, V0 + 2, W[0]);   // 经过射线 P[0] 上的附加点
            dijkstra(V0 + 1, V);
            printf("%lld\n", dist[V0 + 2]);
        } else {
            puts("0");   // 这份代码不处理 k > 2
        }
    }
    return 0;
}

第三档：100 分（+ 区间 DP）

k > 2 时，扇区有 k 个，按环序排好。设扇区 s 的左端射线颜色是 c[s]，右端是 c[s+1]：

  • 如果 c[s] == c[s+1]，这段边界两侧同色，割线不需要穿过它；
  • 如果 c[s] != c[s+1]，这段是"颜色变化处"，割线必须经过它附近。

于是把"颜色变化"的扇区挑出来（数量一定是偶数，因为环上颜色变化的次数必为偶），按环序编号 0..q−1，并记类型：

  • 类型 0：左白右黑（0→1）；
  • 类型 1：左黑右白（1→0）。

割线把这些点两两配对，每对的代价是两个扇区之间的最短路。类型必须相反才能配对（一条割线从"白转黑"处进、从"黑转白"处出）。

为什么可以只考虑"不相交的配对"：如果两条路径交叉，把它们的端点交换（配对 (a,c)、(b,d) 改成 (a,d)、(b,c)），由平面性和三角不等式，总长度不会变大。所以最优解一定可以调成两两不相交。

环上"不相交的完美配对"就是括号匹配：破环为链（把序列复制一遍接在后面），做区间 DP：

文本
f[l][r] = min {
            f[l][k] + f[k+1][r]                       (k = l+1, l+3, …)
            dist[l][r] + f[l+1][r-1]   (若 type[l] != type[r])
          }
答案 = min over l = 0..q-1 of  f[l][l+q-1]

dist 需要对每个变化扇区跑一次 Dijkstra（共 q 次，q ≤ k ≤ 50，且 Σk ≤ 50）。

复杂度 O(Σk · nm log(nm) + Σk³)。

C++
// CSP-S 2021 T4 交通规划 —— 100 分
// 平面图最小割 -> 对偶图最短路；多色时按"颜色变化处"配对 + 区间 DP。
//
// 建模：
//   把原网格的每个小方格看成一个"内部面"(共 (n-1)(m-1) 个)，对偶边权 = 原边权。
//   外部面被 k 个附加点所在的射线切成 k 个"扇区"，扇区 s 从射线 pts[s] 顺时针到 pts[s+1]，
//   它覆盖的那段边界上的每条边界边，都在对偶图里连一条边到边界内侧的内部面。
//   相邻扇区之间隔着一条射线，通过该射线上附加点的边（权 x3）相连。
//   于是"把黑附加点和白附加点分开的最小割" = 对偶图上若干条互不相交路径的长度和。
//
// 求法：
//   只有颜色发生变化的扇区（左端颜色 != 右端颜色）才需要被割线穿过，这样的扇区有偶数个。
//   设它们按环序为 0..q-1，类型 t[i]（0->1 或 1->0）。配对必须"类型相反"，且最优解里
//   各条路径两两不相交 => 合法的配对就是环上的括号匹配 => 破环为链后区间 DP。
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 505;
const int MAXV = MAXN * MAXN + 64;
const long long INF = (long long)1e18;

int n, m, T;
int X1[MAXN][MAXN];   // X1[i][j]: (i,j)-(i+1,j) 竖直边
int X2[MAXN][MAXN];   // X2[i][j]: (i,j)-(i,j+1) 水平边
int V0;               // 内部面个数

struct E { int to, w; };
vector<E> g[MAXV];

inline void addE(int u, int v, int w) {
    g[u].push_back({v, w});
    g[v].push_back({u, w});
}
inline int fid(int i, int j) { return (i - 1) * (m - 1) + j; }   // i in [1,n-1], j in [1,m-1]

int posF[MAXN * 4];   // 每个"位置"对应的内部面（0 = 角点，没有边界边）
int posW[MAXN * 4];

long long dist[MAXV];
int seen[MAXV], seenTag = 0;
typedef pair<long long, int> pli;

void dijkstra(int src, int V) {
    ++seenTag;
    for (int i = 1; i <= V; i++) dist[i] = INF;
    priority_queue<pli, vector<pli>, greater<pli> > pq;
    dist[src] = 0;
    pq.push({0, src});
    while (!pq.empty()) {
        pli t = pq.top(); pq.pop();
        int u = t.second;
        if (seen[u] == seenTag && dist[u] < t.first) continue;
        seen[u] = seenTag;
        for (size_t i = 0; i < g[u].size(); i++) {
            int v = g[u][i].to, w = g[u][i].w;
            if (dist[u] + w < dist[v]) { dist[v] = dist[u] + w; pq.push({dist[v], v}); }
        }
    }
}

int main() {
    if (scanf("%d %d %d", &n, &m, &T) != 3) return 0;
    for (int i = 1; i <= n - 1; i++)
        for (int j = 1; j <= m; j++) scanf("%d", &X1[i][j]);
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m - 1; j++) scanf("%d", &X2[i][j]);

    V0 = (n - 1) * (m - 1);
    // 内部面之间的对偶边
    for (int i = 1; i <= n - 1; i++)
        for (int j = 1; j <= m - 1; j++) {
            if (i + 1 <= n - 1) addE(fid(i, j), fid(i + 1, j), X2[i + 1][j]);
            if (j + 1 <= m - 1) addE(fid(i, j), fid(i, j + 1), X1[i][j + 1]);
        }

    // 射线编号（0-indexed）：上 0..m-1，右 m..m+n-1，下 m+n..2m+n-1，左 2m+n..2m+2n-1
    // 位置 p 表示"射线 p 与射线 p+1 之间"的那段边界
    int N = 2 * m + 2 * n;
    for (int p = 0; p < N; p++) { posF[p] = 0; posW[p] = 0; }
    for (int p = 0; p <= m - 2; p++) { posF[p] = fid(1, p + 1); posW[p] = X2[1][p + 1]; }        // 上
    for (int p = m; p <= m + n - 2; p++) { int i = p - m + 1; posF[p] = fid(i, m - 1); posW[p] = X1[i][m]; } // 右
    for (int p = m + n; p <= 2 * m + n - 2; p++) { int c = 2 * m + n - p; posF[p] = fid(n - 1, c - 1); posW[p] = X2[n][c - 1]; } // 下
    for (int p = 2 * m + n; p <= 2 * m + 2 * n - 2; p++) { int r = 2 * m + 2 * n - p; posF[p] = fid(r - 1, 1); posW[p] = X1[r - 1][1]; } // 左

    vector<int> pp, x3, col;
    vector<int> dynBase;          // 动态结点，每轮清空它们的边
    vector<int> touched;          // 被加过"指向动态结点"的边的内部面，每轮要把这些边删掉
    while (T--) {
        // 清掉上一轮挂在内部面上的动态边
        for (size_t i = 0; i < touched.size(); i++) {
            int f = touched[i];
            size_t w = 0;
            for (size_t j = 0; j < g[f].size(); j++)
                if (g[f][j].to <= V0) g[f][w++] = g[f][j];
            g[f].resize(w);
        }
        touched.clear();
        for (size_t i = 0; i < dynBase.size(); i++) g[dynBase[i]].clear();
        dynBase.clear();

        int k; scanf("%d", &k);
        pp.resize(k); x3.resize(k); col.resize(k);
        for (int i = 0; i < k; i++) {
            scanf("%d %d %d", &x3[i], &pp[i], &col[i]);
            pp[i]--;                // 转 0-indexed
        }
        // 按射线编号排序（环序）
        vector<int> ord(k);
        for (int i = 0; i < k; i++) ord[i] = i;
        sort(ord.begin(), ord.end(), [&](int a, int b) { return pp[a] < pp[b]; });
        vector<int> P(k), C(k), W(k);
        for (int i = 0; i < k; i++) { P[i] = pp[ord[i]]; C[i] = col[ord[i]]; W[i] = x3[ord[i]]; }

        int V = V0 + k;
        for (int s = 0; s < k; s++) { int u = V0 + 1 + s; g[u].clear(); dynBase.push_back(u); }

        // 扇区 s：从射线 P[s] 顺时针到射线 P[(s+1)%k]
        for (int s = 0; s < k; s++) {
            int u = V0 + 1 + s;
            int a = P[s], b = P[(s + 1) % k];
            int t = a;
            while (t != b) {
                if (posF[t]) { addE(u, posF[t], posW[t]); touched.push_back(posF[t]); }
                t = (t + 1) % N;
            }
        }
        // 相邻扇区之间：穿过射线上附加点的边
        for (int i = 0; i < k; i++) {
            int prev = (i - 1 + k) % k;
            addE(V0 + 1 + prev, V0 + 1 + i, W[i]);
        }

        // 收集"颜色变化"的扇区
        vector<int> node;   // 对偶图结点
        vector<int> typ;    // 0: c[s]=0,c[s+1]=1   1: c[s]=1,c[s+1]=0
        for (int s = 0; s < k; s++) {
            int c1 = C[s], c2 = C[(s + 1) % k];
            if (c1 != c2) { node.push_back(V0 + 1 + s); typ.push_back(c1 == 0 ? 0 : 1); }
        }
        int q = (int)node.size();
        if (q == 0) { puts("0"); continue; }

        // 两两最短路
        vector<vector<long long> > D(q, vector<long long>(q, INF));
        for (int i = 0; i < q; i++) {
            dijkstra(node[i], V);
            for (int j = 0; j < q; j++) D[i][j] = dist[node[j]];
        }

        // 环上区间 DP：破环为链，答案取长度 q 的窗口最小值
        int L = 2 * q;
        vector<vector<long long> > f(L + 1, vector<long long>(L + 1, INF));
        for (int len = 2; len <= q; len += 2) {
            for (int l = 0; l + len - 1 < L; l++) {
                int r = l + len - 1;
                long long best = INF;
                if (typ[l % q] != typ[r % q]) {
                    long long inner = (len == 2) ? 0 : f[l + 1][r - 1];
                    if (inner < INF && D[l % q][r % q] < INF) best = min(best, inner + D[l % q][r % q]);
                }
                for (int kk = l + 1; kk < r; kk += 2)
                    if (f[l][kk] < INF && f[kk + 1][r] < INF) best = min(best, f[l][kk] + f[kk + 1][r]);
                f[l][r] = best;
            }
        }
        long long ans = INF;
        for (int l = 0; l < q; l++) ans = min(ans, f[l][l + q - 1]);
        printf("%lld\n", ans);
    }
    return 0;
}

易错点（每一条我都踩过）

  1. 多组询问时动态边没清干净。扇区的边是"临时"的，但 addEdge 会同时在内部面的邻接表里留下指向旧扇区的边。只 clear() 动态结点是不够的 —— 必须把所有被加过边的内部面也处理一遍（把 to > V0 的边删掉）。我一开始只清了动态结点，结果多询问时答案越来越小（n=500、T=25、k=2 时算出 2536，正确是 1572）。
  2. 角点射线重复：2n+2m 条射线里只有 2n+2m−4 个不同的格点，4 个角点被数了两次，对应 4 个"没有边界边"的位置，映射时要跳过。
  3. k = 2 时扇区之间有两条边：一条对应射线 P[0] 上的附加点，一条对应 P[1] 上的。两条都要加（多重边，Dijkstra 自动取较小者）。
  4. q = 0 或 k = 1 直接输出 0：全同色时不需要任何割。
  5. 答案可能很大（路径长度总和 × 106），务必 long long。
  6. dist 数组每次 Dijkstra 都要重置，重置范围是 1..V（V = V0 + k），不要只重置到 V0。
  7. 区间 DP 的下标是"复制两倍后的序列"，type 和 dist 要取模 q。
