2. club 社团招新

2.1 题意与关键结论

n 个人（n 为偶数）分到 3 个部门，要求每个部门人数不超过 n/2，第 i 人放进部门 di 的满意度是 ai,di，最大化总满意度。多组数据。

关键结论（全题唯一的突破口）：三个部门人数之和恒为 n，所以

存在部门人数 > n2 ⟺ 某个部门人数 > n2

即最多只有一个部门会超限。这条结论把"三个容量约束"退化成了"一个约束"，也是后面贪心能成立的原因。

2.2 档位 1：O(n3) 暴力 DP（55 分，测试点 1~11，n≤ 200）

只用记录部门 1、部门 2 各放了几个人（部门 3 的人数 = 已放人数 − 两者之和，自动确定）。

  • 状态：dp[j][k] = 前 i 个人中，部门 1 有 j 人、部门 2 有 k 人时的最大满意度；
  • 转移：第 i 个人三个去向各试一次，同时检查三个容量（j+1≤ n/2、k+1≤ n/2、部门 3 人数 +1≤ n/2）；
  • 复杂度 O(n3)，n=200 时约 8× 106，非常稳。
C++
// CSP-S 2025 T1 社团招新 (club) —— 档1：O(n^3) 暴力 DP
// 适用：测试点 1~11（n<=30 及 n=200），共 55 分
// 编译：g++-15 -O2 -std=c++14 club.cpp -o club
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll NEG = LLONG_MIN / 4;

int main() {
    int T;
    if (scanf("%d", &T) != 1) return 0;
    while (T--) {
        int n; scanf("%d", &n);
        int half = n / 2;
        // dp[j][k]：已处理若干人，其中 j 人在部门1、k 人在部门2（其余在部门3）
        static ll dp[2][205][205];
        for (int j = 0; j <= half; j++)
            for (int k = 0; k <= half; k++) dp[0][j][k] = NEG;
        dp[0][0][0] = 0;
        int cur = 0;
        for (int i = 1; i <= n; i++) {
            ll a[3]; scanf("%lld %lld %lld", &a[0], &a[1], &a[2]);
            for (int j = 0; j <= half; j++)
                for (int k = 0; k <= half; k++) dp[cur ^ 1][j][k] = NEG;
            for (int j = 0; j <= half; j++)
                for (int k = 0; k <= half; k++) {
                    ll v = dp[cur][j][k];
                    if (v == NEG) continue;
                    int l = (i - 1) - j - k;          // 部门3人数
                    if (l < 0 || l > half) continue;
                    if (j + 1 <= half) dp[cur ^ 1][j + 1][k] = max(dp[cur ^ 1][j + 1][k], v + a[0]);
                    if (k + 1 <= half) dp[cur ^ 1][j][k + 1] = max(dp[cur ^ 1][j][k + 1], v + a[1]);
                    if (l + 1 <= half) dp[cur ^ 1][j][k]     = max(dp[cur ^ 1][j][k],     v + a[2]);
                }
            cur ^= 1;
        }
        ll ans = 0;
        for (int j = 0; j <= half; j++)
            for (int k = 0; k <= half; k++) {
                int l = n - j - k;
                if (l < 0 || l > half) continue;
                if (dp[cur][j][k] != NEG) ans = max(ans, dp[cur][j][k]);
            }
        printf("%lld\n", ans);
    }
    return 0;
}

为什么只有 55 分：n=105 时 O(n3) 完全不可能，第 12 个点就过不去了。它只能吃掉"n≤ 200 全包"的 11 个点。

2.3 档位 2：特殊性质 A（ai,2=ai,3=0）（60 分，测试点 12）

部门 2、3 的满意度恒为 0，放进去就是白给。所以：

  • 只有部门 1 能产生收益，容量是 n/2；
  • 于是把 ai,1 最大的 n/2 个人放进部门 1，其余随便丢进部门 2/3（收益 0）。

答案 = ai,1 中最大的 n/2 个之和。排序一次即可，O(nlog n)。

C++
// CSP-S 2025 T1 社团招新 (club) —— 档2：特殊性质 A (a[i][2]=a[i][3]=0)
// 适用：测试点 12，累计 60 分
// 结论：只有部门1有分，把 a[i][1] 最大的 n/2 个人放进部门1，其余随便丢（贡献 0）。
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int T; scanf("%d", &T);
    while (T--) {
        int n; scanf("%d", &n);
        int half = n / 2;
        vector<ll> v;
        for (int i = 1; i <= n; i++) {
            ll a, b, c; scanf("%lld %lld %lld", &a, &b, &c);
            v.push_back(a);                 // 只用第一列
        }
        sort(v.begin(), v.end(), greater<ll>());
        ll ans = 0;
        for (int i = 0; i < half; i++) ans += v[i];
        printf("%lld\n", ans);
    }
    return 0;
}

2.4 档位 3：特殊性质 B（ai,3=0）（70 分，测试点 9,13,14）

部门 3 恒为 0，所以没有人主动想去部门 3；又因为部门 1、2 的人数之和必须是 n、各自 ≤ n/2，两个部门都恰好是 n/2 人。

于是问题变成："选 n/2 个人进部门 1，其余进部门 2"。

  • 先假设所有人都在部门 2：base=∑i ai,2；
  • 把第 i 个人从部门 2 挪到部门 1 的净收益是 ai,1−ai,2；
  • 每个人独立、没有别的约束，所以取净收益最大的 n/2 个即可。

答案 =base+∑ 最大的 n/2 个 (ai,1−ai,2)。同样 O(nlog n)。

C++
// CSP-S 2025 T1 社团招新 (club) —— 档3：特殊性质 B (a[i][3]=0)
// 适用：测试点 13,14，累计 70 分
// 结论：a[i][3]=0 时最优解里 n 个人必然全被分完（部门1、2 各恰好 n/2 人）。
//       答案 = sum a[i][2] + (a[i][1]-a[i][2]) 中最大的 n/2 个之和。
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int T; scanf("%d", &T);
    while (T--) {
        int n; scanf("%d", &n);
        int half = n / 2;
        vector<ll> d;
        ll base = 0;
        for (int i = 1; i <= n; i++) {
            ll a, b, c; scanf("%lld %lld %lld", &a, &b, &c);
            base += b;                     // 先假设所有人都去部门2
            d.push_back(a - b);            // 改成去部门1 的“增量”
        }
        sort(d.begin(), d.end(), greater<ll>());
        for (int i = 0; i < half; i++) base += d[i];
        printf("%lld\n", base);
    }
    return 0;
}

2.5 档位 4：正解 —— 后悔贪心（100 分）

第一步：先不管容量，让每个人去自己最满意的部门。

记 mx 为最优部门、best = a[i][mx]、second = 三个数里的次大值。这一步给出的是上界。

第二步：由 2.1 的结论，至多一个部门 d 超限。

设 cnt[d] > n/2，需要从 d 里"赶出去" need = cnt[d] - n/2 个人。

第三步：赶人的代价。

把一个原本在 d 的人挪走，他最好的去处是另外两个部门中更好的那个，所以代价是

lossi = besti − secondi ≥ 0

为了让总满意度损失最小，把 d 里的人按 loss 升序排序，挪走最小的 need 个。

为什么这样是对的：

  1. 答案的上界就是"无容量约束时的最大和"，而最优解必须是它的某个"改动"；
  2. 只有 d 超限，所以任何可行解都必须把 d 里至少 need 个人挪走；
  3. 挪走一个非 d 部门的人不但不减 d 的人数，还会破坏总收益，永远不优；
  4. 挪走的人之间互不影响（各自独立地选自己第二好的部门），所以选 loss 最小的 need 个是贪心最优；
  5. 搬完之后：d 恰好 n/2 人，另外两个部门合计 n/2 人，因此各自都不超过 n/2，合法。
C++
// CSP-S 2025 T1 社团招新 (club) —— 正解：反悔贪心  O(n log n)
// 适用：全部测试点，100 分
//
// 引理1：按“每人先去自己最满意的部门”得到的三个部门人数 c1,c2,c3，
//        至多只有一个部门 > n/2（因为三者之和为 n）。
// 引理2：若部门 d 超载 e = c_d - n/2 人，只需从“最满意为 d”的人中挑 e 个改去自己的次优部门，
//        且一定不会让别的部门再次超载（因为另两部门的前缀和 <= n/2 - c_others）。
// 引理3：为了让损失最小，按 (最优值 - 次优值) 升序取前 e 个。
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int T; scanf("%d", &T);
    while (T--) {
        int n; scanf("%d", &n);
        int half = n / 2;
        long long base = 0;
        int cnt[3] = {0, 0, 0};
        vector<ll> loss[3];
        for (int i = 1; i <= n; i++) {
            ll a[3];
            scanf("%lld %lld %lld", &a[0], &a[1], &a[2]);
            int mx = 0;                            // 满意度最高的部门
            if (a[1] > a[mx]) mx = 1;
            if (a[2] > a[mx]) mx = 2;
            ll second = 0;                         // 其余两个部门中的较大值
            for (int j = 0; j < 3; j++)
                if (j != mx) second = max(second, a[j]);
            base += a[mx];
            cnt[mx]++;
            loss[mx].push_back(a[mx] - second);    // 改去次优部门的损失
        }
        for (int d = 0; d < 3; d++) {
            if (cnt[d] > half) {
                sort(loss[d].begin(), loss[d].end());
                int need = cnt[d] - half;
                for (int i = 0; i < need; i++) base -= loss[d][i];
            }
        }
        printf("%lld\n", base);
    }
    return 0;
}

复杂度 O(nlog n)（只对超限部门的 loss 排序），n=105 实测 0.05 s / 8.8 MB。

3. road 道路修复

3.1 题意

n 个点、m 条原图乡村道路（带边权）。另有 k 个乡镇，第 j 个乡镇的"改造费"为 cj；若把该乡镇改造，则它可以向点 i 修一条边权为 aj,i 的路。

自由选择要改造的乡镇集合 S，此时可用的边集是

原图边 ∪ {aj,i | j∈ S, 1≤ i≤ n}

要求图连通，最小化

（生成树的边权和） + ∑j∈ S cj

先想清楚一件事：改造费是"买路权"的一次性开销，与选多少条边无关。所以枚举 S（2k 种）后，问题就是"在这个边集上求 MST"，再在枚举里取 min。

3.2 贯穿全题的三个观察

  1. 每个被改造的乡镇最多贡献 n−1 条边（剩下的边必然被 MST 淘汰）。所以任何合法方案只需在 O(n + kn) 条边上做 MST，不必理会 m 条原图边里的"废边"。
  2. 原图里"不可能进 MST"的边，对任何 S 都不可能进 MST——因为它们连在原图上的权值就已经劣了。因此先对原图单独跑一次 Kruskal，只留下 n−1 条树边，其余 m−(n−1) 条可以永久扔掉。
  3. 枚举 S 时不要重复排序：把"要用的边"一次性排好序，之后每个 S 只做一遍线性扫描。

第 3 点是 L2 → L3 的分水岭，第 2 点是 L3 → 正解的分水岭。

3.3 L1：k=0（16 分，测试点 1~4）

k=0，一个乡镇都不能改造，答案就是对原图跑 Kruskal 的 MST 边权和。O(mlog m)。

C++
// CSP-S 2025 T2 道路修复 (road) —— 档1 (预期 16 分, 测试点 1~4: k=0)
// 只有原图 m 条边, 直接跑一遍 Kruskal 输出即可。
// 注意：k>0 时它忽略乡镇, 答案会偏大 —— 这正是它只有 16 分的原因。

3.4 L2：2k 枚举 + 每个集合重新排序（32 分，测试点 1~8）

最直接的做法：枚举 mask，把"原图 m 条边 + 已改造乡镇的边"合起来排序，跑一遍 Kruskal，取最小值。

  • m≤ 105, k≤ 5 时：25× 105log ≈ 5× 107，能过；
  • m=106 时：32× 106× 20 ≈ 6× 108，直接超时——所以它只能吃到第 8 个点。
C++
// CSP-S 2025 T2 道路修复 (road) —— 档2 (预期 32 分, 测试点 1~8: m<=1e5, k<=5)
// 枚举 2^k 个改造集合, 每个集合把「原图 m 条边 + 选中乡镇的 k*n 条边」合并后重新排序跑 Kruskal。

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
static inline int rd() {
    int x = 0; char ch = getchar();
    while (ch < '0' || ch > '9') ch = getchar();
    while (ch >= '0' && ch <= '9') { x = x * 10 + (ch - '0'); ch = getchar(); }
    return x;

    orig.resize(m);
    for (int i = 0; i < m; i++) { orig[i].u = rd(); orig[i].v = rd(); orig[i].w = rd(); }
    townEdge.assign(k + 1, {});
    for (int j = 1; j <= k; j++) {
        c[j] = rd();
        for (int i = 1; i <= n; i++) townEdge[j].push_back({n + j, i, rd()});
    }
    ll ans = LLONG_MAX;
    vector<Edge> cur;
    for (int mask = 0; mask < (1 << k); mask++) {
        cur = orig;                                     // 每次重新拷贝 m 条边
        ll cost = 0;
        for (int j = 1; j <= k; j++)
            if (mask >> (j - 1) & 1) { cost += c[j]; cur.insert(cur.end(), townEdge[j].begin(), townEdge[j].end()); }
        sort(cur.begin(), cur.end(), [](const Edge& a, const Edge& b) { return a.w < b.w; });
        for (int i = 1; i <= n + k; i++) fa[i] = i;
        for (const Edge& e : cur) {
            int u = find(e.u), v = find(e.v);
            if (u != v) { fa[u] = v; cost += e.w; }
        }
        ans = min(ans, cost);
    }
    printf("%lld\n", ans);
    return 0;
}

3.5 L3：全边一次性预排序 + 跳过未改造乡镇（64 分，所有 k≤ 5 的点）

把原来的"每个 mask 排一次序"改成"所有 mask 共用同一个有序数组"：

  1. 取出原图的 m 条边 加上 k 个乡镇的全部边（共 m+kn 条），按边权一次性排序；
  2. 枚举 mask，对排序后的数组做一次线性扫描跑 Kruskal：遇到 town != 0 && mask 中该镇未被改造 的边就 continue；
  3. 加剪枝 if (cost >= ans) continue;。

复杂度 O(mlog m + 2k(n+kn))。

实测：n=104, m=106, k=5 用时 0.29 s，答案与正解完全一致；但 k=10 时要 2.04 s（210× (n+kn) 太大），所以止步在 k≤ 5 的 16 个点。

// 把排序提到枚举外面：所有 m + k*n 条边只排一次, 枚举集合时线性扫描并跳过未改造乡镇的边。

3.6 正解：原图只留 MST 树边 + 预排序扫描（100 分）

在 L3 的基础上加观察 2：

  • 对原图单独跑一次 Kruskal，得到 n−1 条树边；其余 m−n+1 条边永久丢弃；
  • 把"这 n−1 条树边 + kn 条城乡边"一次性排序，作为所有 mask 共用的数组；
  • 每个 mask 只在这个 O(n+kn) 的数组上线性扫描跑 Kruskal。

复杂度降到

O(mlog m + (n+kn)log(n+kn) + 2k(n+kn))

n=104, m=106, k=10 实测 0.65 s，卡在 1 s 之内。

C++
// CSP-S 2025 T2 道路修复 (road) —— 正解 (100 分)
// 关键两步：
//   1) 原图先跑一遍 Kruskal，只保留 n-1 条树边：任何一条非树边在任何改造方案下都不会被选中。
//   2) 把这 n-1 条树边 + 所有 k*n 条城乡边一次性排序，枚举 2^k 个改造集合时只做线性扫描，
//      跳过“没被改造的乡镇”所对应的边。
// 复杂度 O(m log m + (nk) log(nk) + 2^k * (n + k) * α)
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

struct Edge { int u, v, w, town; };   // town=0 表示城市边, 否则为 1..k

static inline int rd() {
    int x = 0; char c = getchar();
    while (c < '0' || c > '9') c = getchar();
    while (c >= '0' && c <= '9') { x = x * 10 + (c - '0'); c = getchar(); }
    return x;
}

int n, m, k;
vector<Edge> base;
vector<int> c;                       // 改造费用
int fa[10015];
int find(int x) { while (fa[x] != x) { fa[x] = fa[fa[x]]; x = fa[x]; } return x; }

int main() {
    n = rd(); m = rd(); k = rd();
    vector<Edge> orig(m);
    for (int i = 0; i < m; i++) {
        int u = rd(), v = rd(), w = rd();
        orig[i] = {u, v, w, 0};
    }
    // ---- 第 1 步：原图 Kruskal，只留树边 ----
    sort(orig.begin(), orig.end(), [](const Edge& a, const Edge& b) { return a.w < b.w; });
    for (int i = 1; i <= n; i++) fa[i] = i;
    base.reserve(n - 1 + (size_t)k * n);
    for (const Edge& e : orig) {
        int u = find(e.u), v = find(e.v);
        if (u != v) { fa[u] = v; base.push_back(e); }
    }
    // ---- 读入乡镇, 生成城乡边 ----
    c.assign(k + 1, 0);
    for (int j = 1; j <= k; j++) {
        c[j] = rd();
        for (int i = 1; i <= n; i++) base.push_back({n + j, i, rd(), j});
    }
    // ---- 第 2 步：一次性排序 ----
    sort(base.begin(), base.end(), [](const Edge& a, const Edge& b) { return a.w < b.w; });

    ll ans = LLONG_MAX;
    for (int mask = 0; mask < (1 << k); mask++) {
        for (int i = 1; i <= n + k; i++) fa[i] = i;
        ll cost = 0;
        for (int j = 1; j <= k; j++) if (mask >> (j - 1) & 1) cost += c[j];
        if (cost >= ans) continue;                    // 小剪枝
        for (const Edge& e : base) {
            if (e.town && !(mask >> (e.town - 1) & 1)) continue;
            int u = find(e.u), v = find(e.v);
            if (u != v) { fa[u] = v; cost += e.w; }
        }
        if (cost < ans) ans = cost;
    }
    printf("%lld\n", ans);
    return 0;
}

3.7 实现要点与坑

  • 改造费一定要计入：total = MST 边权和 + Σ_{j∈S} c_j，很多人的第一版会忘掉 cj。
  • 并查集重置：210=1024 次 Kruskal，每次 memset 整个 n 数组即可（104× 1024 = 107，可接受）；也可以用"时间戳版本号"把重置降到 O(1) 每次。
  • 边权与答案要开 long long：边权 109、n=104，MST 边权和可达 1013。
  • 枚举时的剪枝顺序：先算改造费，cost >= ans 就 continue，能省掉大量无用扫描。
  • 性质 A（cj=0 且存在 aj,i=0）：改造是免费的，全部改造（或至少把所有 cj=0 的乡镇都改造）不会让答案变差，可以退化成"一次 Kruskal"，是 L1/L2 档位在 5,6,9,10 号点上的捷径。

4. replace 谐音替换

4.1 题意

给出 n 个二元组 (si,1,si,2)，满足 |si,1|=|si,2|。对字符串 s 的一次替换是：挑一个子串 y=si,1（写成 s=x+y+z），把它换成 y'=si,2，得到 x+si,2+z。

q 个询问 (tj,1,tj,2)：问有多少种替换能把 tj,1 变成 tj,2。两种替换不同 ⟺ 子串位置不同 或 用到的二元组不同。

数据：n,q≤ 2×105，L1=∑(|si,1|+|si,2|)≤ 5×106，L2 同理。

4.2 第一步：把"替换"翻译成"窗口"

因为 |si,1|=|si,2|，替换不改变字符串长度，所以先决条件是 |tj,1|=|tj,2|。

一次替换 (i, 位置 p) 能成立 ⟺ 在 t1 上开一个长度 Li 的窗口 [p, p+Li−1]，使得

  1. t1[p..p+Li−1]=si,1；
  2. t2[p..p+Li−1]=si,2；
  3. 窗口外面的 t1、t2 完全相同。

设 lo、hi 是 t1,t2 第一个 / 最后一个不同的位置。由于窗口外的部分必须逐字符相同，条件 3 等价于

[lo, hi]⊆ [p, p+Li−1] ⟺ p≤ lo, p+Li−1≥ hi

于是合法的 p 范围是

p∈[max(0, hi−Li+1), min(lo, |t1|−Li)]

条件 3 是最容易漏掉的一条——只判断"同一位置两串同时匹配"会把样例 1 算成 3（正确答案是 2）。

4.3 档位 1：按定义暴力（10 分，测试点 1,2）

枚举 (i,p)，直接用 std::string 逐字符比较两个窗口。复杂度 O(nL2) 量级，L≤200 时可过。

C++
// T3 参考暴力（对拍基准）：严格按定义 —— t1 = x + s1_i + z 且 t2 = x + s2_i + z
#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, q;
    if (scanf("%d %d", &n, &q) != 2) return 0;
    vector<string> s1(n), s2(n);
    static char buf[5000006];
    for (int i = 0; i < n; i++) {
        scanf("%s", buf); s1[i] = buf;
        scanf("%s", buf); s2[i] = buf;
    }
    while (q--) {
        scanf("%s", buf); string t1 = buf;
        scanf("%s", buf); string t2 = buf;
        long long ans = 0;
        if (t1.size() == t2.size()) {
            int L1 = (int)t1.size();
            // 两串不同的“最小包围区间” [lo, hi]
            int lo = -1, hi = -1;
            for (int i = 0; i < L1; i++) if (t1[i] != t2[i]) { if (lo < 0) lo = i; hi = i; }
            if (lo >= 0) {
                for (int i = 0; i < n; i++) {
                    int L = (int)s1[i].size();
                    for (int p = 0; p + L <= L1; p++) {
                        if (p > lo || p + L - 1 < hi) continue;      // 窗口必须盖住差异区
                        if (t1.compare(p, L, s1[i]) == 0 && t2.compare(p, L, s2[i]) == 0) ans++;
                    }
                }
            }
        }
        printf("%lld\n", ans);
    }
    return 0;
}

4.4 档位 2：哈希暴力（50 分，测试点 1~8、13,14）

把"逐字符比较"换成"O(1) 的哈希比较"：

  1. 预处理每条规则 (si,1,si,2) 的整体哈希值；
  2. 每次询问先 O(L2) 扫一遍求出 lo,hi，并建好 t1,t2 的前缀哈希；
  3. 枚举 i，再枚举 p∈[max(0,hi−Li+1),min(lo,|t1|−Li)]，O(1) 判两个窗口哈希。

每次询问复杂度 O(L2+∑i|si,1|)（与 q 无关地与所有规则过一遍）。于是：

  • 测试点 1~6：n,q≤103、L≤2000，直接稳过；
  • 测试点 7,8 和 13,14：它们的 L 上限高达 106∼5×106，但特殊性质 A 保证 q=1，整个程序只跑一次询问，代价就是 O(L1/2+L2)，照样秒过；
  • 测试点 11,12、15~20：q 到了 2×105，q× L1 直接爆掉，必须上正解。

所以这个"看起来只能拿 25 分"的暴力，实际靠 q=1 能拿到 50 分——这是这道题最值钱的一个部分分。

C++
// CSP-S 2025 T3 谐音替换 (replace) —— 档1：暴力 (预期 25 分, 测试点 1~5)
// 定义：答案 = #{(i, p) : t1 = x + s1_i + z 且 t2 = x + s2_i + z}
// 等价条件（设 [lo,hi] 为 t1,t2 第一个/最后一个不同位置）：
//   (1) 窗口 [p, p+L-1] 覆盖 [lo,hi]，即 p <= lo 且 p+L-1 >= hi；
//   (2) t1[p..p+L-1] = s1_i；(3) t2[p..p+L-1] = s2_i。
// 复杂度 O(n * L2)，n 和 L2 都在 1e3~2e3 量级时可过。
#include <bits/stdc++.h>
using namespace std;
typedef unsigned long long ull;
const ull B = 131;
static char b1[5000006], b2[5000006];
static ull h1[5000006], h2[5000006], pw[5000006];

int main() {
    int n, q;
    scanf("%d %d", &n, &q);
    vector<int> len(n);
    vector<ull> H1(n), H2(n);
    for (int i = 0; i < n; i++) {
        scanf("%s %s", b1, b2);
        int m = (int)strlen(b1);
        len[i] = m;
        ull x = 0, y = 0;
        for (int j = 0; j < m; j++) { x = x * B + (ull)(b1[j] - 'a' + 1); y = y * B + (ull)(b2[j] - 'a' + 1); }
        H1[i] = x; H2[i] = y;
    }
    pw[0] = 1;
    for (int i = 1; i < 5000006; i++) pw[i] = pw[i - 1] * B;
    for (int qi = 0; qi < q; qi++) {
        scanf("%s %s", b1, b2);
        int L1 = (int)strlen(b1), L2 = (int)strlen(b2);
        long long ans = 0;
        if (L1 == L2) {
            h1[0] = 0; for (int j = 0; j < L1; j++) h1[j + 1] = h1[j] * B + (ull)(b1[j] - 'a' + 1);
            h2[0] = 0; for (int j = 0; j < L2; j++) h2[j + 1] = h2[j] * B + (ull)(b2[j] - 'a' + 1);
            int lo = -1, hi = -1;
            for (int j = 0; j < L1; j++) if (b1[j] != b2[j]) { if (lo < 0) lo = j; hi = j; }
            if (lo >= 0) {
                for (int i = 0; i < n; i++) {
                    int L = len[i];
                    int pmin = max(0, hi - L + 1), pmax = min(lo, L1 - L);
                    for (int p = pmin; p <= pmax; p++) {
                        if (h1[p + L] - h1[p] * pw[L] == H1[i] && h2[p + L] - h2[p] * pw[L] == H2[i]) ans++;
                    }
                }
            }
        }
        printf("%lld\n", ans);
    }
    return 0;
}

4.5 正解：拆 (A,B,C) / (AA,BQ,CC) + 分组 + 两棵 Trie + 差分 BIT（100 分）

4.5.1 结论

把规则拆成

si,1=Ai+Bi+Ci, si,2=Ai+Di+Ci

其中 Ai = 两串的最长公共前缀，Ci = 最长公共后缀，Bi、Di 是剩下的中间段（长度相等）。

询问同样拆成

tj,1=AAj+BQj+CCj, tj,2=AAj+DQj+CCj

（注意 lo=|AAj|，所以 BQj=t1[lo..hi]、DQj=t2[lo..hi]。）

那么——

规则 i 对询问 j 的贡献只可能是 0 或 1；贡献为 1 的充要条件是

Bi=BQj, Di=DQj, Ai 是 AAj 的后缀, Ci 是 CCj 的前缀

4.5.2 为什么

  • 替换位置 p 由"中间段对齐"唯一夹定：p=|AAj|−|Ai|。所以匹配与否只有 0/1，不会出现同一个规则有多个合法 p。
  • 窗口左端 [p, p+|Ai|−1] 的 A 段必须落在 t1,t2 相同的区域里，否则会成为"不同的字符"，于是 Ai 必须是 AAj 的后缀；
  • 窗口右端的 C 段同理必须落在公共后缀里，所以 Ci 必须是 CCj 的前缀；
  • 反过来，若 |Bi|≠|BQj|，窗口的 A 段或 C 段必然跨过 lo / hi 边界，那一段就不可能同时在 t1,t2 里相同。所以只有 B,D 完全相同才可能（A、C 可以被"吸收"进 AA、CC 的更长部分里，B、D 不能）。

4.5.3 怎么数

按 (B,D) 分组（key 用字符串 B + '#' + D）。组内要数的是

#{i: Ai 是 AAj 的后缀, Ci 是 CCj 的前缀}

两个"前后缀关系"都能翻译成 Trie 上的祖先关系：

条件 等价说法 Trie 上的含义
Ai 是 AAj 的后缀 rev(Ai) 是 rev(AAj) 的前缀 nodeA(rev(A_i)) 是 nodeA(rev(AA_j)) 的祖先
Ci 是 CCj 的前缀 同义 nodeC(C_i) 是 nodeC(CC_j) 的祖先

于是用经典套路：

  1. 建 TrieA（插所有 rev(Ai)）和 TrieC（插所有 Ci），各自求 Euler 序 tin/tout；
  2. 在 TrieA 上 DFS：进入结点 u 时，把"nodeA 恰好是 u 的那些规则"在 TrieC 上做子树区间加 [tinC, toutC] += 1；离开 u 时撤销（-= 1）；
  3. 走到某个询问所在的结点时，BIT 里的 +1 恰好对应"nodeA 是它祖先（或自身）"的规则；此时单点查 tinC(nodeC(CC_j))，得到的就是"nodeC(C_i) 是它的祖先"的规则个数。
  4. 区间加 + 单点查用差分树状数组。

每个 (B,D) 分组单独建一次 TrieA/TrieC 并跑一次 DFS+BIT，跑完清零复用。

4.5.4 复杂度与实测

每组 Trie 的规模是该组规则的 |Ai|+|Ci|，所有组求和为 O(L1)；询问侧就是 walk 一遍字符串 O(L2)。因此

O(L1+L2+(n+q)log L1) 时间, O(L1) 空间

实测 n=105, q=5×104 用时 0.06 s；构造"长公共前后缀"的压力数据 0.12 s / 79 MB。

// CSP-S 2025 T3 谐音替换 (replace) —— 正解  O(L1 + L2 + (n+q) log)
// ---------------------------------------------------------------------------
// 关键转化（用定义 t1 = x + s1_i + z, t2 = x + s2_i + z 推出）
//  规则 i : s1 = A + B + C,  s2 = A + D + C   (A=最长公共前缀, C=最长公共后缀, B/D=中间段)
//  询问 j : t1 = AA + BQ + CC, t2 = AA + DQ + CC  (同样取最长公共前后缀)
//  规则 i 对询问 j 恰好贡献 1 条，当且仅当
//      B == BQ  且  D == DQ  且  A 是 AA 的后缀  且  C 是 CC 的前缀
//  （替换位置被两端唯一夹定，所以匹配的规则至多贡献 1。）
// ---------------------------------------------------------------------------
// 计数：按 (B,D) 分组。组内要数 #{ i : A_i 是 AA 的后缀, C_i 是 CC 的前缀 }。
//  · A 是 AA 的后缀 <=> rev(A) 是 rev(AA) 的前缀
//  · C 是 CC 的前缀 <=> C 是 CC 的前缀
// 两者都是「前缀」关系 —— 在 Trie 上就是「结点的祖先关系」：
//  记 nodeA(x)=x 在 TrieA 中走到的结点（走不动就停在最深处，根 = 空串）。
//  要数的是 #{ i : nodeA(revA_i) 是 nodeA(revAA_j) 的祖先 且 nodeC(C_i) 是 nodeC(CC_j) 的祖先 }。
// 经典技巧：在 TrieA 上 DFS，进入结点 u 时把「nodeA=u 的规则 i」在 TrieC 上做
//  区间加 [tinC(C_i), toutC(C_i)] += 1，离开时撤销；到询问所在结点时，BIT 里恰好是
//  「nodeA 是祖先」的那些规则，单点查 tinC(nodeC(CC_j)) 得到的就是
//  「nodeC(C_i) 是它的祖先」的个数。区间加 + 单点查用差分 BIT。
#include <bits/stdc++.h>
using namespace std;

/* ---------------- 快速输入 ---------------- */
static char ibuf[1 << 25];
static size_t ipos = 0, ilen = 0;
static inline int gc() { if (ipos == ilen) { ilen = fread(ibuf, 1, sizeof(ibuf), stdin); ipos = 0; if (!ilen) return -1; } return ibuf[ipos++]; }
static inline void readStr(string& s) {
    s.clear(); int c = gc();
    while (c != -1 && (c == ' ' || c == '\n' || c == '\r' || c == '\t')) c = gc();
    while (c != -1 && c != ' ' && c != '\n' && c != '\r' && c != '\t') { s.push_back((char)c); c = gc(); }
}
static inline int readInt() { int c = gc(); while (c < '0' || c > '9') c = gc(); int x = 0; while (c >= '0' && c <= '9') { x = x * 10 + c - '0'; c = gc(); } return x; }

/* ---------------- Trie（前向星孩子） ---------------- */
struct Trie {
    vector<int> head, nxt, to; vector<unsigned char> ch; int cnt = 0;
    void reset() { head.clear(); nxt.clear(); to.clear(); ch.clear(); cnt = 0; newNode(); }
    inline int newNode() { head.push_back(-1); return cnt++; }
    inline int getChild(int u, int c) {
        for (int e = head[u]; e != -1; e = nxt[e]) if (ch[e] == (unsigned char)c) return to[e];
        int v = newNode(); ch.push_back((unsigned char)c); to.push_back(v); nxt.push_back(head[u]); head[u] = (int)to.size() - 1; return v;
    }
    inline int findChild(int u, int c) {
        for (int e = head[u]; e != -1; e = nxt[e]) if (ch[e] == (unsigned char)c) return to[e];
        return -1;
    }
    inline int walk(const string& s) { int u = 0; for (char c : s) { int v = findChild(u, c); if (v == -1) break; u = v; } return u; }
};

/* ---------------- 差分 BIT：区间加 + 单点查 ---------------- */
static vector<long long> bitv; static int bitn = 0;
static inline void bitUpd(int i, long long v) { for (++i; i <= bitn; i += i & (-i)) bitv[i] += v; }
static inline void rangeAdd(int l, int r, long long v) { bitUpd(l, v); if (r + 1 < bitn) bitUpd(r + 1, -v); }
static inline long long pointGet(int i) { long long s = 0; for (++i; i > 0; i -= i & (-i)) s += bitv[i]; return s; }

/* ---------------- 迭代 DFS 求 tin / tout ---------------- */
static void euler(Trie& T, vector<int>& tin, vector<int>& tout) {
    int N = T.cnt; tin.assign(N, 0); tout.assign(N, 0);
    vector<int> stN, stE; int timer = 0;
    stN.push_back(0); tin[0] = timer++;
    stE.push_back(T.head[0]);
    while (!stN.empty()) {
        int u = stN.back(); int& e = stE.back();
        if (e == -1) { tout[u] = timer - 1; stN.pop_back(); stE.pop_back(); continue; }
        int v = T.to[e]; e = T.nxt[e];
        tin[v] = timer++;
        stN.push_back(v); stE.push_back(T.head[v]);
    }
}

int main() {
    int n = readInt(), q = readInt();
    unordered_map<string, int> gmap;
    gmap.reserve((size_t)n * 2 + 10);
    vector<string> ruleA, ruleC;                     // rev(A) 与 C
    vector<vector<int>> gRules, gQueries;

    string s1, s2;
    for (int i = 0; i < n; i++) {
        readStr(s1); readStr(s2);
        int len = (int)s1.size();
        int a = 0; while (a < len && s1[a] == s2[a]) a++;
        if (a == len) continue;                      // s1 == s2 不产生贡献
        int b = 0; while (b < len - a && s1[len - 1 - b] == s2[len - 1 - b]) b++;
        int blen = len - a - b;
        string key; key.reserve(2 * blen + 1);
        key.append(s1, a, blen); key.push_back('#'); key.append(s2, a, blen);
        auto it = gmap.find(key);
        int gid;
        if (it == gmap.end()) { gid = (int)gRules.size(); gmap.emplace(std::move(key), gid); gRules.push_back({}); gQueries.push_back({}); }
        else gid = it->second;
        string revA(s1.begin(), s1.begin() + a); reverse(revA.begin(), revA.end());
        string C(s1.begin() + len - b, s1.end());
        gRules[gid].push_back((int)ruleA.size());
        ruleA.push_back(std::move(revA));
        ruleC.push_back(std::move(C));
    }

    vector<int> qgid(q, -1), qNodeA(q, 0), qNodeC(q, 0);
    vector<string> qRevAA(q), qCC(q);
    for (int j = 0; j < q; j++) {
        readStr(s1); readStr(s2);
        if (s1.size() != s2.size()) continue;
        int L = (int)s1.size(), lo = -1, hi = -1;
        for (int t = 0; t < L; t++) if (s1[t] != s2[t]) { if (lo < 0) lo = t; hi = t; }
        if (lo < 0) continue;                        // t1 == t2
        string key; key.reserve(2 * (hi - lo + 1) + 1);
        key.append(s1, lo, hi - lo + 1); key.push_back('#'); key.append(s2, lo, hi - lo + 1);
        auto it = gmap.find(key);
        if (it == gmap.end()) continue;              // 该分组不存在 => 答案 0
        qgid[j] = it->second;
        gQueries[it->second].push_back(j);
        string revAA(s1.begin(), s1.begin() + lo); reverse(revAA.begin(), revAA.end());
        qRevAA[j] = std::move(revAA);
        qCC[j] = s1.substr(hi + 1);
    }

    vector<long long> ans(q, 0);
    Trie TA, TC;
    vector<int> tinA, toutA, tinC, toutC;
    vector<int> atA, nxtA, valA, atC, nxtC, valC;
    vector<int> qAtA, qNxtA, qValA;
    vector<int> nodeCof(ruleA.size(), 0);

    for (int g = 0; g < (int)gRules.size(); g++) {
        if (gQueries[g].empty()) continue;
        int m = (int)gRules[g].size();
        (void)m;
        /* ---------- 建 TrieA（rev(A)） ---------- */
        TA.reset();
        atA.assign(1, -1); nxtA.clear(); valA.clear();
        for (int id : gRules[g]) {
            int u = 0;
            for (char c : ruleA[id]) { u = TA.getChild(u, c); if ((int)atA.size() < TA.cnt) atA.resize(TA.cnt, -1); }
            if ((int)atA.size() < TA.cnt) atA.resize(TA.cnt, -1);
            valA.push_back(id); nxtA.push_back(atA[u]); atA[u] = (int)valA.size() - 1;
        }
        atA.resize(TA.cnt, -1);
        euler(TA, tinA, toutA);
        /* ---------- 建 TrieC（C） ---------- */
        TC.reset();
        atC.assign(1, -1); nxtC.clear(); valC.clear();
        for (int id : gRules[g]) {
            int u = 0;
            for (char c : ruleC[id]) { u = TC.getChild(u, c); if ((int)atC.size() < TC.cnt) atC.resize(TC.cnt, -1); }
            if ((int)atC.size() < TC.cnt) atC.resize(TC.cnt, -1);
            nodeCof[id] = u;
            valC.push_back(id); nxtC.push_back(atC[u]); atC[u] = (int)valC.size() - 1;
        }
        atC.resize(TC.cnt, -1);
        euler(TC, tinC, toutC);
        /* ---------- 询问各自走到的最深结点 ---------- */
        qAtA.assign(TA.cnt, -1); qNxtA.clear(); qValA.clear();
        for (int j : gQueries[g]) {
            qNodeA[j] = TA.walk(qRevAA[j]);
            qNodeC[j] = TC.walk(qCC[j]);
            qValA.push_back(j); qNxtA.push_back(qAtA[qNodeA[j]]); qAtA[qNodeA[j]] = (int)qValA.size() - 1;
        }
        /* ---------- TrieA 上 DFS + BIT ---------- */
        bitn = TC.cnt; bitv.assign(bitn + 1, 0);
        vector<int> stN, stE;
        // 进入结点 u：加上 nodeA=u 的规则（在 TrieC 上做子树区间加），再回答挂在 u 上的询问
        auto enterNode = [&](int u) {
            for (int e = atA[u]; e != -1; e = nxtA[e]) { int id = valA[e]; rangeAdd(tinC[nodeCof[id]], toutC[nodeCof[id]], +1); }
            for (int e = qAtA[u]; e != -1; e = qNxtA[e]) { int j = qValA[e]; ans[j] = pointGet(tinC[qNodeC[j]]); }
        };
        auto leaveNode = [&](int u) {
            for (int e = atA[u]; e != -1; e = nxtA[e]) { int id = valA[e]; rangeAdd(tinC[nodeCof[id]], toutC[nodeCof[id]], -1); }
        };
        enterNode(0);
        stN.push_back(0); stE.push_back(TA.head[0]);
        while (!stN.empty()) {
            int u = stN.back(); int& e = stE.back();
            if (e == -1) { leaveNode(u); stN.pop_back(); stE.pop_back(); continue; }
            int v = TA.to[e]; e = TA.nxt[e];
            stN.push_back(v); stE.push_back(TA.head[v]);
            enterNode(v);
        }
    }
    string out;
    for (int j = 0; j < q; j++) { out += to_string(ans[j]); out.push_back('\n'); }
    fwrite(out.data(), 1, out.size(), stdout);
    return 0;
}

4.6 这一题的几个致命坑

  1. 漏"窗口外必须相同"：这是本场最容易挂的地方，样例 1 就会暴露（错算成 3）。
  2. 分组 key 别用 (A,B,C,D)：只有 B,D 必须完全相同，A,C 是可以被更长的 AA,CC 吸收的。key 用 B#D，询问侧就是 t1[lo..hi]#t2[lo..hi]。
  3. Trie 动态开点后立刻 resize 附属数组：getChild 可能新建结点，忘记 resize 会读越界/丢数据。
  4. 输入要快读：L1+L2 可达 107 字符，cin >> string 会慢到超时，必须 fread 手写缓冲。
  5. 逐组重建、别跨组复用：TrieA/TrieC 与 BIT 必须按分组清空，否则不同 (B,D) 的规则会串味。

5. employ 员工招聘

5.1 题意

n 天，每天面试一个人；小 Z 决定面试顺序 p（1∼ n 的排列）。第 i 天的题目难度 si∈{0,1}：si=0 表示没人做得出（拒绝），si=1 表示所有人都做得出（录用）。

编号 i 的人耐心为 ci：若在他之前已经有"不少于 ci 人"被拒绝或放弃参加面试，那么他也放弃参加面试。

求能录用至少 m 人的排列 p 的数量（对 998244353 取模）。

1≤ m≤ n≤ 500，si∈{0,1}，0≤ ci≤ n。

5.2 把过程形式化（全题的地基）

设 ai 表示第 i 天的人是否被录用，bi 表示前 i−1 天里"未录用（被拒 ∪ 放弃）"的人数。若第 i 天来的人耐心为 c，则：

情形 结果 约束
si=0 必然不被录用 对 c 无限制
si=1 且不录用 他放弃了 c≤ bi（上限）
si=1 且录用 他做了题 c>bi（下限）

记最终录用 k 人，则未录用 j=n−k 人，"录用至少 m 人" ⟺ j≤ n−m。这个 j 就是下面 DP 的第一维。

整道题就是"在一个同时有上限约束和下限约束的排列计数问题"。

5.3 档位 1：全排列暴力（8 分，测试点 1,2）

next_permutation 枚举 p，按 5.2 的三条规则模拟即可。O(n!· n)，n≤10 时 10!×10≈ 3.6×107，能过。

C++
// CSP-S 2025 T4 employ —— 档位1：全排列暴力
// 覆盖测试点 1,2 (n<=10)  预期 8 分
// 复杂度 O(n! * n)
#include <bits/stdc++.h>
using namespace std;
int n, m;
char s[32];
int c[32];
int main() {
    scanf("%d %d", &n, &m);
    scanf("%s", s + 1);
    for (int i = 1; i <= n; i++) scanf("%d", &c[i]);

    vector<int> p(n);
    iota(p.begin(), p.end(), 1);   // p[j] = 第 j+1 天来面试的人
    long long ans = 0;
    do {
        int bad = 0, hire = 0;
        for (int i = 1; i <= n; i++) {
            int u = p[i - 1];
            if (bad >= c[u]) {
                // 之前已有 >= c_u 人被拒绝或放弃 -> 他放弃
                bad++;
            } else if (s[i] == '0') {
                // 参加但做不出题 -> 被拒绝
                bad++;
            } else {
                // 做出题 -> 录用
                hire++;
            }
        }
        if (hire >= m) ans++;
    } while (next_permutation(p.begin(), p.end()));

    printf("%lld\n", ans);
    return 0;
}

5.4 档位 2：状压 DP（20 分，测试点 3~5）

注意一个关键恒等式：未录用数 = 已过天数 − 录用数，所以只要记住"面试了哪些人"和"录用了几个"，未录用数就算出来了。

  • 状态 dp[mask][h]：已面试集合为 mask，其中录用 h 人；
  • 已过天数 i=popcount(mask)+1，未录用数 bad=popcount(mask)−h；
  • 转移：枚举一个未面试的人 u，按 5.2 三条规则决定 h 变不变。

复杂度 O(2n n2)，n=18 时约 2×107，实测 0.04 s。

C++
// CSP-S 2025 T4 employ —— 档位2：状压 DP
// 覆盖测试点 3~5 (n<=18)  累计 20 分
// dp[mask][h] : 已面试集合 mask，其中录用 h 人
//   由 mask 可得已过天数 i=popcount(mask)，未录用(=被拒/放弃)人数 bad = popcount(mask)-h
// 复杂度 O(2^n * n^2)，内存 O(2^n * n)
#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
int n, m;
char s[32];
int c[32];

int main() {
    scanf("%d %d", &n, &m);
    scanf("%s", s + 1);
    for (int i = 1; i <= n; i++) scanf("%d", &c[i]);

    int full = 1 << n;
    vector<int> dp((size_t)full * (n + 1), 0);
    auto at = [&](int mask, int h) -> int& { return dp[(size_t)mask * (n + 1) + h]; };
    at(0, 0) = 1;

    for (int mask = 0; mask < full; mask++) {
        int pc = __builtin_popcount(mask);
        int day = pc + 1;                 // 下一天
        if (day > n) continue;
        char sd = s[day];
        for (int h = 0; h <= pc; h++) {
            int cur = dp[(size_t)mask * (n + 1) + h];
            if (!cur) continue;
            int bad = pc - h;
            for (int u = 1; u <= n; u++) {
                if (mask >> (u - 1) & 1) continue;
                int nh;
                if (bad >= c[u]) nh = h;          // 放弃
                else if (sd == '0') nh = h;       // 参加但被拒
                else nh = h + 1;                  // 录用
                int &to = dp[(size_t)(mask | (1 << (u - 1))) * (n + 1) + nh];
                to += cur;
                if (to >= MOD) to -= MOD;
            }
        }
    }

    long long ans = 0;
    for (int h = m; h <= n; h++) ans += at(full - 1, h);
    printf("%lld\n", ans % MOD);
    return 0;
}

5.5 档位 3：m=1 专用（测试点 12~14）

答案 = n! −（一个人都没录用的排列数）。

"一个人都没录用" ⟺ 每天来的人都成为"未录用"，也就是第 i 天之前的未录用数恒为 i−1：

  • si=0：这天的人必然未录用，对 c 无约束；
  • si=1：这天的人必须放弃，即要求 c≤ i−1。

设 S=#{i:si=1}，把这些天按 i 升序记为 i1<…<iS。先给这些受约束的位置选人：

#{c≤ it−1}−(t−1)

（前 t−1 个位置用掉的人耐心都 ≤ it−1−1≤ it−1，所以都在同一个集合里，直接减。）剩下 n−S 个位置全是 s=0、无约束，自由排剩余的人 ×(n−S)!。

ans=n!−∏t=1S(dit−1−(t−1))×(n−S)!

O(nlog n)。

C++
// CSP-S 2025 T4 employ —— 档位3：m=1 专用（测试点 12~14，n<=500, m=1）
// 思路：答案 = n! - (一个人都没录用的方案数)
//   "一个人都没录用" 等价于：每一天来的人都成为"未录用"（被拒或放弃），即第 i 天之前未录用数恒为 i-1。
//   - s_i = 0：该天的人必然未录用，对 c 无限制；
//   - s_i = 1：该天的人必须"放弃"，即要求 c <= i-1。
//   记 S = "1" 的个数，把 s_i=1 的天按 i 升序记为 i_1<...<i_S。
//   先给这些受约束的位置选人：第 t 个位置可选 #{c<=i_t-1} - (t-1) 人；
//   剩下 n-S 个（全是 s=0）位置自由排剩余的人，乘 (n-S)!。
//   本文件只在 m=1 时正确。
#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
int n, m;
char s[505];
int cnt[505], d[505];
long long fac[505];

int main() {
    scanf("%d %d", &n, &m);
    scanf("%s", s + 1);
    for (int i = 1; i <= n; i++) { int x; scanf("%d", &x); cnt[x]++; }
    for (int i = 0; i <= n; i++) d[i] = (i ? d[i - 1] : 0) + cnt[i];   // d[i]=#{c<=i}
    // d[-1] 视为 0：用 d[i-1] 时 i>=1，i=1 时 d[0]=cnt[0]，注意这里表示 #{c<=i-1}=d[i-1]
    fac[0] = 1;
    for (int i = 1; i <= n; i++) fac[i] = fac[i - 1] * i % MOD;

    int S = 0;
    for (int i = 1; i <= n; i++) if (s[i] == '1') S++;

    long long zero = 1;     // 0 人录用的方案数
    int t = 0;
    for (int i = 1; i <= n; i++) {
        if (s[i] != '1') continue;
        t++;
        long long avail = (i - 1 >= 0 ? d[i - 1] : 0) - (t - 1);   // #{c <= i-1} 中还没用的
        if (avail <= 0) { zero = 0; break; }
        zero = zero * (avail % MOD) % MOD;
    }
    zero = zero * fac[n - S] % MOD;

    long long ans = (fac[n] - zero) % MOD;
    if (ans < 0) ans += MOD;
    printf("%lld\n", ans);
    return 0;
}

5.6 档位 4：m=n 专用（测试点 15）

要录用全部 n 人，必须每天都能录用：

  • 只要有某个 si=0，第 i 天的人必然被拒 → 答案 0；
  • 于是 s 必须全为 1。此时只要所有 ci≥1，未录用数永远是 0，而 0<ci 恒成立，就没人会放弃，每个人的题都做得出来 → n! 种排列全部合法；
  • 若存在 ci=0：该人轮到时"未录用数 0≥0"必然放弃 → 答案 0。

ans={n!,s 全为 1 且所有 ci≥10,否则

C++
// CSP-S 2025 T4 employ —— 档位4：m=n 专用（测试点 15，n<=500, m=n）
// 思路：要录用全部 n 人，必须每一天都成功录用。
//   - 只要有某个 s_i = 0，第 i 天的人必然被拒 -> 不可能，答案 0；
//   - 于是 s 全为 1；此时只要所有 c_i >= 1，就永远不会有人放弃
//     （未录用数恒为 0 < c_i，每个人都参加面试且做出题），全部 n! 种排列都合法；
//   - 若存在 c_i = 0，该人轮到时"未录用数 0 >= 0"必然放弃 -> 答案 0。
// 结论：answer = n!  当且仅当 (s 全为 '1' 且 min c_i >= 1)，否则 0。
#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
int main() {
    int n, m;
    static char s[505];
    scanf("%d %d", &n, &m);
    scanf("%s", s + 1);
    int mnc = INT_MAX;
    for (int i = 1; i <= n; i++) { int x; scanf("%d", &x); mnc = min(mnc, x); }
    bool ok = (mnc >= 1);
    for (int i = 1; i <= n; i++) if (s[i] != '1') ok = false;
    long long ans = 0;
    if (ok) {
        ans = 1;
        for (int i = 1; i <= n; i++) ans = ans * i % MOD;
    }
    printf("%lld\n", ans);
    return 0;
}

5.7 档位 5：特殊性质 A（所有 si=1）（测试点 6~8、16,17）

性质 A 不需要新算法：s 全为 1 时，正解 DP 里 s='0' 那条分支永远不会被走到，代码只是正解去掉一个分支。所以"先写出正解框架"就能直接吃掉 6~8（n≤100）和 16,17（n≤500）这 20 分。

C++
// CSP-S 2025 T4 employ —— 档位3：特殊性质 A 专版（测试点 6~8、16,17；所有 s_i = 1）
// 说明：性质 A 不需要新算法，正解 DP 在 s 全为 1 时只剩"录用/不录用"三个转移。
//       本文件即把 std 中 s=0 的分支删去的版本（仅当 s 全为 1 时保证正确）。
#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
const int N = 505;
int n, m;
char s[N];
int cnt[N], d[N];
int f[N][N], g[N][N];
long long fac[N];

int main() {
    scanf("%d %d", &n, &m);
    scanf("%s", s + 1);
    for (int i = 1; i <= n; i++) { int x; scanf("%d", &x); cnt[x]++; }
    d[0] = cnt[0];
    for (int i = 1; i <= n; i++) d[i] = d[i - 1] + cnt[i];
    fac[0] = 1;
    for (int i = 1; i <= n; i++) fac[i] = fac[i - 1] * i % MOD;

    f[0][0] = 1;
    for (int i = 1; i <= n; i++) {          // s[i] 恒为 '1'
        memset(g, 0, sizeof(g));
        for (int j = 0; j <= n; j++)
            for (int k = 0; k <= n; k++) {
                int cur = f[j][k];
                if (!cur) continue;
                int avail = d[j] - k; if (avail < 0) avail = 0;
                long long w = (long long)cur * avail % MOD;
                int &a1 = g[j + 1][k + 1]; a1 += (int)w; if (a1 >= MOD) a1 -= MOD;   // 不录用：真上限
                int &a2 = g[j][k];         a2 += cur;   if (a2 >= MOD) a2 -= MOD;   // 录用，不钦定
                int &a3 = g[j][k + 1];     a3 -= (int)w; if (a3 < 0) a3 += MOD;     // 录用但钦定违反
            }
        memcpy(f, g, sizeof(f));
    }
    long long ans = 0;
    for (int j = 0; j <= n - m; j++)
        for (int k = 0; k <= n; k++)
            ans = (ans + (long long)f[j][k] * fac[n - k]) % MOD;
    printf("%lld\n", ans);
    return 0;
}

5.8 档位 6：特殊性质 B（∑i si≤18）（测试点 18~21）

一个必须一眼看出来的剪枝：录用只可能发生在 si=1 的那天，所以

录用人数 ≤ S:=∑i si≤ 18

  • 若 m>S，直接输出 0（这一条能挡掉不少点）；
  • 反向地，未录用人数 j≥ n−S，所以正解 DP 的答案求和下标从 n−S 起。

其余完全复用正解。n≤500 时正解本来就是 0.09 s 级别，加上剪枝更快。

5.9 正解：容斥 + 贡献延后 DP（100 分）

5.9.1 先解决"只有上限"的版本

如果所有约束都是"c≤某上限"，答案是现成的：

把 n 个上限从小到大排成 up1≤ up2≤⋯≤ upn，则方案数

∏t=1n(cntupt−(t−1)), cntx=#{i:ci≤ x}

（第 t 个位置（上限最小）只能从 cntupt 个"够小"的人里挑，前面已经用掉 t−1 个。）

本题里 bi 随 i 单调不降，所以上限天然就是从小到大排好的，不需要额外排序。

5.9.2 用容斥把"下限"变成"上限"，并延后贡献

三类约束里，si=1 且录用那种是下限（c>bi）。对它做容斥：

#{c>bi}=#{任意}不钦定−#{c≤ bi}钦定违反，系数 −1

也就是：把"录用位"先当成"没有任何约束"，同时额外允许它"被钦定成上限 c≤ bi、带 −1 系数"。这样一来，全部约束都变成上限了。

贡献延后：不钦定的招募位，当下不乘任何系数（因为还不知道有多少人可选），留到全部决策做完之后，统一按"给剩下 n−k 个自由位置随便分配剩下的人"来处理，也就是最后乘 (n−k)!。

5.9.3 DP 转移（本场的核心）

设 dj=#{i:ci≤ j}，f[j][k] = 处理到当前位置，未录用 j 人、已确定 k 个"上限位"（真上限 + 钦定违反下限的）的方案数。

当前 si 选择 转移到 系数
'0' 不录用 f[j+1][k] +f[j][k]
'1' 不录用（真上限 c≤ j） f[j+1][k+1] +f[j][k]·(dj−k)
'1' 录用，不钦定（无约束） f[j][k] +f[j][k]
'1' 录用，但钦定违反下限（假上限 c≤ j） f[j][k+1] −f[j][k]·(dj−k)

(dj−k) 的含义：所有"上限位"的人耐心都 ≤ 当前 j（因为 b 单调不降），已经用掉 k 个，所以还能从 cntj−k 个里挑一个。

最终答案

ans=∑j=0n−m ∑k=0n f[j][k]×(n−k)!

(n−k)! 就是"给剩下 n−k 个没有上限约束的位置自由分配剩余人"的方案数（等价于把这些位置的上限当成 n）。

5.9.4 为什么这个容斥表达的是"c>bi"

只看最后一步，设当前 j,k：

(n−k)!−（n−k−1)!·(dj−k)=(n−k−1)![(n−k)−(dj−k)]=(n−k−1)!·(n−dj)

右式的 n−dj 正是"c>j 的人数"——也就是录用位真正可以挑的人。"任意 − 上限"这一步恰好把下限 c>j 还原了出来，这就是这个 DP 成立的原因。

5.9.5 复杂度

O(n3) 时间（i× j× k，n=500 时 1.25×108），O(n2) 空间。 实测 n=500 最坏 0.09 s / 8.9 MB，相对 1 s / 512 MiB 有极大余量。

C++
// CSP-S 2025 T4 employ —— 档位5：正解（容斥 + 贡献延后 DP）
// 覆盖全部测试点 1~25  n<=500  100 分
// 复杂度 O(n^3)，内存 O(n^2)
//
// 思路：
//   设 a_i 表示第 i 天的人是否被录用，b_i 表示前 i-1 天里"未录用(被拒/放弃)"的人数。
//   第 i 天的人耐心为 c 时：
//     s_i=0              -> 必不录用                    （对 c 无限制）
//     s_i=1 且不录用      -> 放弃，要求 c <= b_i        （上限约束）
//     s_i=1 且录用        -> 参加并录用，要求 c > b_i   （下限约束）
//   "只有上限"的排列计数：把每个位置的上限排成非降序列 up_1..up_n，
//    方案数 = prod (cnt_{up_t} - (t-1))，cnt_x = 人数中 c<=x 的个数。
//   下限用容斥去掉：钦定某些"录用"位违反下限（改成上限 c<=b_i），系数 -1。
//   DP: f[j][k] = 处理到当前位置，b=j（未录用 j 人），已确定 k 个"上限位"（含真上限与钦定违反的）方案数。
//     s_i=0 : f[j+1][k]   += f[j][k]
//     s_i=1 : f[j+1][k+1] += f[j][k] * (cnt_j - k)      // 不录用（真上限 c<=j）
//             f[j  ][k  ] += f[j][k]                    // 录用，且不钦定（下限留到末尾一起算）
//             f[j  ][k+1] -= f[j][k] * (cnt_j - k)      // 录用但钦定违反下限（容斥，负号）
//   答案 = sum_{j=0}^{n-m} sum_k f[j][k] * (n-k)!
//   其中 (n-k)! 是给"没有上限约束的 n-k 个位置"自由分配剩余人的方案数（把它们的上限视作 n）。
#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
const int N = 505;

int n, m;
char s[N];
int cnt[N], d[N];
int f[N][N], g[N][N];
long long fac[N];

int main() {
    scanf("%d %d", &n, &m);
    scanf("%s", s + 1);
    for (int i = 1; i <= n; i++) { int x; scanf("%d", &x); cnt[x]++; }
    d[0] = cnt[0];
    for (int i = 1; i <= n; i++) d[i] = d[i - 1] + cnt[i];   // d[i] = #{c<=i}

    fac[0] = 1;
    for (int i = 1; i <= n; i++) fac[i] = fac[i - 1] * i % MOD;

    f[0][0] = 1;
    for (int i = 1; i <= n; i++) {
        memset(g, 0, sizeof(g));
        for (int j = 0; j <= n; j++)
            for (int k = 0; k <= n; k++) {
                int cur = f[j][k];
                if (!cur) continue;
                if (s[i] == '0') {
                    int &to = g[j + 1][k];
                    to += cur; if (to >= MOD) to -= MOD;
                } else {
                    int avail = d[j] - k;
                    if (avail < 0) avail = 0;
                    long long w = (long long)cur * avail % MOD;   // 真正可选的 c<=j 的人数
                    // 1) 不录用：真上限，系数 +1
                    int &a1 = g[j + 1][k + 1];
                    a1 += (int)w; if (a1 >= MOD) a1 -= MOD;
                    // 2) 录用且不钦定：无约束
                    int &a2 = g[j][k];
                    a2 += cur; if (a2 >= MOD) a2 -= MOD;
                    // 3) 录用但钦定"违反下限"（等于强行给它一个上限 c<=j），系数 -1
                    int &a3 = g[j][k + 1];
                    a3 -= (int)w; if (a3 < 0) a3 += MOD;
                }
            }
        memcpy(f, g, sizeof(f));
    }

    long long ans = 0;
    for (int j = 0; j <= n - m; j++)
        for (int k = 0; k <= n; k++)
            ans = (ans + (long long)f[j][k] * fac[n - k]) % MOD;
    printf("%lld\n", ans);
    return 0;
}

5.10 踩坑清单

  1. 容斥项是负号：g[j][k+1] -= w，取模时必须先 +MOD，否则出负数。
  2. (dj−k) 可能为负（不可达状态），要 max(0, ...)，否则会算进垃圾贡献。
  3. 答案是所有 j≤ n−m 的和，不是 j=n−m 单点——题目要的是"至少 m 人"。
  4. 别忘 b 单调不降这个前提：(dj−k) 能 O(1) 算出来全靠它。
  5. 输入用 long long 参与乘法；int 存中间乘积会溢出。
